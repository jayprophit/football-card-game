-- FootballChess Database Schema
-- Complete database structure for the game with user management, cards, matches, and blockchain integration

-- Users and Authentication
CREATE TABLE users (
  user_id UUID PRIMARY KEY,
  username VARCHAR(50) UNIQUE NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  date_of_birth DATE NOT NULL,
  age INT NOT NULL,
  is_underage BOOLEAN NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  last_login TIMESTAMP,
  country_code VARCHAR(3),
  language_code VARCHAR(5),
  is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE user_profiles (
  profile_id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  display_name VARCHAR(50),
  avatar_url TEXT,
  bio TEXT,
  favorite_team VARCHAR(100),
  preferred_formation VARCHAR(20),
  total_matches INT DEFAULT 0,
  wins INT DEFAULT 0,
  draws INT DEFAULT 0,
  losses INT DEFAULT 0,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE user_sessions (
  session_id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  token VARCHAR(255) NOT NULL,
  ip_address VARCHAR(45),
  user_agent TEXT,
  is_valid BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  expires_at TIMESTAMP NOT NULL
);

-- Age Verification and KYC
CREATE TABLE age_verification (
  verification_id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  date_of_birth DATE NOT NULL,
  age INT NOT NULL,
  is_underage BOOLEAN NOT NULL,
  verified BOOLEAN DEFAULT FALSE,
  verification_method VARCHAR(50),
  verification_date TIMESTAMP,
  unlock_date TIMESTAMP,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE kyc_requests (
  request_id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  status VARCHAR(20) NOT NULL,
  submitted_at TIMESTAMP NOT NULL DEFAULT NOW(),
  processed_at TIMESTAMP,
  rejection_reason TEXT,
  documents_count INT NOT NULL
);

CREATE TABLE kyc_documents (
  document_id UUID PRIMARY KEY,
  request_id UUID REFERENCES kyc_requests(request_id) ON DELETE CASCADE,
  document_type VARCHAR(50) NOT NULL,
  document_hash VARCHAR(255) NOT NULL,

-- Card System
CREATE TABLE card_templates (
  template_id VARCHAR(50) PRIMARY KEY,
  position VARCHAR(20) NOT NULL,
  movement_type VARCHAR(20) NOT NULL,
  base_rating INT NOT NULL,
  rarity VARCHAR(20) NOT NULL,
  image_url TEXT,
  created_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE player_cards (
  card_id UUID PRIMARY KEY,
  template_id VARCHAR(50) REFERENCES card_templates(template_id),
  position VARCHAR(20) NOT NULL,
  rating INT NOT NULL,
  number INT NOT NULL,
  movement_type VARCHAR(20) NOT NULL,
  speed INT NOT NULL,
  passing INT NOT NULL,
  shooting INT NOT NULL,
  defending INT NOT NULL,
  durability INT NOT NULL,
  is_legendary BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE user_cards (
  user_card_id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  card_id UUID REFERENCES player_cards(card_id) ON DELETE CASCADE,
  experience INT DEFAULT 0,
  matches_played INT DEFAULT 0,
  nft_token_id VARCHAR(100),
  is_in_team BOOLEAN DEFAULT FALSE,
  acquired_at TIMESTAMP NOT NULL DEFAULT NOW(),
  nft_status VARCHAR(20) DEFAULT 'not_minted',
  last_upgraded_at TIMESTAMP
);

CREATE TABLE custom_cards (
  custom_card_id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  card_id UUID REFERENCES player_cards(card_id) ON DELETE CASCADE,
  original_image_url TEXT,
  processed_image_url TEXT,
  created_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE card_packs (
  pack_id UUID PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  description TEXT,
  price DECIMAL(10,2) NOT NULL,
  cards_count INT NOT NULL,
  rarity_distribution JSONB,
  is_available BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE user_pack_purchases (
  purchase_id UUID PRIMARY KEY,

-- Team System
CREATE TABLE teams (
  team_id UUID PRIMARY KEY,

  name VARCHAR(100) NOT NULL,
  formation VARCHAR(20) NOT NULL,

  updated_at TIMESTAMP NOT NULL DEFAULT NOW()

CREATE TABLE team_cards (
  team_id UUID REFERENCES teams(team_id) ON DELETE CASCADE,
  user_card_id UUID REFERENCES user_cards(user_card_id) ON DELETE CASCADE,
  position_index INT NOT NULL,

);

CREATE TABLE formations (

-- Match System
CREATE TABLE matches (
  match_id UUID PRIMARY KEY,
  home_team_id UUID REFERENCES teams(team_id) ON DELETE SET NULL,

  turn_count INT DEFAULT 0,
  current_turn VARCHAR(20),
  tournament_id UUID,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  started_at TIMESTAMP,
  ended_at TIMESTAMP,
  match_data JSONB
);

CREATE TABLE match_events (
  event_id UUID PRIMARY KEY,

  event_type VARCHAR(20) NOT NULL,

  timestamp TIMESTAMP NOT NULL DEFAULT NOW(),

);

CREATE TABLE match_moves (
  move_id UUID PRIMARY KEY,
  match_id UUID REFERENCES matches(match_id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  card_id UUID,

  from_y INT NOT NULL,

  has_ball BOOLEAN DEFAULT FALSE,

-- Create view for leaderboard
CREATE VIEW user_leaderboard AS
SELECT 
  u.user_id,
  u.username,
  up.display_name,
  up.avatar_url,
  pp.level,
  pp.experience,
  pp.matches_played,
  pp.matches_won,
  pp.matches_drawn,
  pp.matches_lost,
  pp.goals_scored,
  pp.goals_conceded,
  pp.matches_won * 3 + pp.matches_drawn AS points,
  CASE WHEN pp.matches_played > 0 THEN 
    CAST(pp.matches_won AS FLOAT) / pp.matches_played * 100 
  ELSE 0 END AS win_percentage
FROM 
  users u
JOIN 
  user_profiles up ON u.user_id = up.user_id
JOIN 
  player_progression pp ON u.user_id = pp.user_id
WHERE 
  u.is_active = TRUE;

  ml.item_type,
  ml.item_id,
  ml.price,
  ml.currency,
  ml.status,
  ml.created_at,
  ml.expires_at,
  CASE 
    WHEN ml.item_type = 'player_card' THEN 
      jsonb_build_object(
        'card_id', pc.card_id,
        'position', pc.position,
        'rating', pc.rating,
        'is_legendary', pc.is_legendary
      )
    WHEN ml.item_type = 'team' THEN
      jsonb_build_object(
        'team_id', t.team_id,
        'name', t.name,
        'formation', t.formation,
        'rating', t.rating
      )
    ELSE NULL
  END AS item_details
FROM 
  marketplace_listings ml
JOIN 
  users u ON ml.seller_id = u.user_id
LEFT JOIN 
  player_cards pc ON ml.item_type = 'player_card' AND ml.item_id::UUID = pc.card_id
LEFT JOIN 
  teams t ON ml.item_type = 'team' AND ml.item_id::UUID = t.team_id
WHERE 
  ml.status = 'active';

-- Create functions for age verification
CREATE OR REPLACE FUNCTION update_user_age()
RETURNS TRIGGER AS $
BEGIN
  -- Calculate age
  NEW.age := EXTRACT(YEAR FROM AGE(CURRENT_DATE, NEW.date_of_birth));
  -- Set underage flag
  NEW.is_underage := NEW.age < 18;
  RETURN NEW;
END;
$ LANGUAGE plpgsql;

CREATE TRIGGER before_user_insert_update
BEFORE INSERT OR UPDATE OF date_of_birth ON users
FOR EACH ROW
EXECUTE FUNCTION update_user_age();

-- Create function to update age verification statuses
CREATE OR REPLACE FUNCTION update_age_verifications()
RETURNS VOID AS $
DECLARE
  r RECORD;
BEGIN
  FOR r IN 
    SELECT 
      av.user_id, 
      av.date_of_birth,
      av.unlock_date,
      EXTRACT(YEAR FROM AGE(CURRENT_DATE, av.date_of_birth)) AS new_age
    FROM 
      age_verification av
    WHERE 
      av.is_underage = TRUE AND
      EXTRACT(YEAR FROM AGE(CURRENT_DATE, av.date_of_birth)) >= 18
  LOOP
    -- Update age verification record
    UPDATE age_verification 
    SET 
      age = r.new_age,
      is_underage = FALSE,
      updated_at = NOW()
    WHERE 
      user_id = r.user_id;
    
    -- Update user record
    UPDATE users
    SET
      age = r.new_age,
      is_underage = FALSE
    WHERE
      user_id = r.user_id;
      
    -- Trigger notification
    INSERT INTO notifications (
      notification_id,
      user_id,
      type,
      title,
      message,
      action_data
    ) VALUES (
      uuid_generate_v4(),
      r.user_id,
      'age_verification',
      'Age Restriction Lifted',
      'You are now eligible for all features in FootballChess!',
      jsonb_build_object('unlockDate', r.unlock_date)
    );
  END LOOP;
END;
$ LANGUAGE plpgsql;

-- Create function to process pending NFTs
CREATE OR REPLACE FUNCTION check_nft_eligibility()
RETURNS VOID AS $
DECLARE
  r RECORD;
BEGIN
  FOR r IN 
    SELECT 
      uc.user_card_id,
      uc.user_id,
      uc.card_id,
      u.age,
      u.is_underage,
      k.status AS kyc_status
    FROM 
      user_cards uc
    JOIN 
      users u ON uc.user_id = u.user_id
    LEFT JOIN 
      kyc_requests k ON u.user_id = k.user_id AND k.status = 'approved'
    WHERE 
      uc.nft_status = 'pending_age' AND
      u.is_underage = FALSE
  LOOP
    -- Update NFT status to pending_kyc if user is not underage but KYC not approved
    IF r.kyc_status IS NULL THEN
      UPDATE user_cards
      SET nft_status = 'pending_kyc'
      WHERE user_card_id = r.user_card_id;
    ELSE
      -- Update NFT status to pending_mint if user is not underage and KYC is approved
      UPDATE user_cards
      SET nft_status = 'pending_mint'
      WHERE user_card_id = r.user_card_id;
      
      -- Create notification
      INSERT INTO notifications (
        notification_id,
        user_id,
        type,
        title,
        message,
        action_data
      ) VALUES (
        uuid_generate_v4(),
        r.user_id,
        'nft_eligible',
        'NFT Minting Available',
        'Your player card is now eligible for minting as an NFT!',
        jsonb_build_object('cardId', r.card_id)
      );
    END IF;
  END LOOP;
END;
$ LANGUAGE plpgsql;
