// FootballChess Encrypted Messaging System
// Provides secure, end-to-end encrypted communication between players

import * as nacl from 'tweetnacl';
import * as naclUtil from 'tweetnacl-util';
import * as uuid from 'uuid';
import * as socketio from 'socket.io-client';
import { createHash } from 'crypto';

/**
 * Encrypted Messaging System
 * Handles secure communications between players, team chat, and trade negotiations
 */
class EncryptedMessagingSystem {
  constructor(config) {
    this.config = config;
    this.socket = null;
    this.keyPair = null;
    this.userID = null;
    this.publicKeys = new Map(); // Store public keys of other users
    this.conversations = new Map(); // Store conversation metadata
    this.messages = new Map(); // Store messages by conversation ID
    this.listeners = new Map(); // Message listeners
    this.connected = false;
    this.messageCounter = 0; // For nonce generation
  }
  
  /**
   * Initialize the messaging system
   * @param {string} userID User ID
   * @returns {Promise<void>}
   */
  async initialize(userID) {
    this.userID = userID;
    
    // Generate key pair
    this.keyPair = nacl.box.keyPair();
    
    // Connect to messaging server
    await this._connectToServer();
    
    // Register public key with server
    await this._registerPublicKey();
    
    // Set up message handlers
    this._setupMessageHandlers();
    
    // Load existing conversations
    await this._loadExistingConversations();
    
    console.log('Encrypted messaging system initialized');
  }
  
  /**
   * Connect to messaging server
   * @returns {Promise<void>}
   */
  async _connectToServer() {
    return new Promise((resolve, reject) => {
      try {
        this.socket = socketio.connect(this.config.serverUrl, {
          query: { userID: this.userID }
        });
        
        this.socket.on('connect', () => {
          this.connected = true;
          console.log('Connected to messaging server');
          resolve();
        });
        
        this.socket.on('connect_error', (error) => {
          console.error('Connection error:', error);
          reject(error);
        });
        
        this.socket.on('disconnect', () => {
          this.connected = false;
          console.log('Disconnected from messaging server');
        });
      } catch (error) {
        console.error('Socket connection error:', error);
        reject(error);
      }
    });
  }
  
  /**
   * Register public key with server
   * @returns {Promise<void>}
   */
  async _registerPublicKey() {
    return new Promise((resolve, reject) => {
      if (!this.connected) {
        reject(new Error('Not connected to server'));
        return;
      }
      
      const publicKeyBase64 = naclUtil.encodeBase64(this.keyPair.publicKey);
      
      this.socket.emit('register_public_key', {
        userID: this.userID,
        publicKey: publicKeyBase64
      }, (response) => {
        if (response.success) {
          console.log('Public key registered successfully');
          resolve();
        } else {
          console.error('Public key registration failed:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Set up message handlers
   */
  _setupMessageHandlers() {
    // Handle incoming encrypted messages
    this.socket.on('encrypted_message', async (data) => {
      try {
        // Decrypt the message
        const decryptedMessage = await this._decryptMessage(data);
        
        // Store the message
        this._storeMessage(decryptedMessage);
        
        // Trigger message listeners
        this._triggerMessageListeners(decryptedMessage);
      } catch (error) {
        console.error('Error handling encrypted message:', error);
      }
    });
    
    // Handle public key updates
    this.socket.on('public_key_update', (data) => {
      const { userID, publicKey } = data;
      this.publicKeys.set(userID, naclUtil.decodeBase64(publicKey));
      console.log(`Updated public key for user ${userID}`);
    });
    
    // Handle conversation updates
    this.socket.on('conversation_update', (data) => {
      this._handleConversationUpdate(data);
    });
  }
  
  /**
   * Load existing conversations
   * @returns {Promise<void>}
   */
  async _loadExistingConversations() {
    return new Promise((resolve, reject) => {
      this.socket.emit('get_conversations', { userID: this.userID }, (response) => {
        if (response.success) {
          response.conversations.forEach(conversation => {
            this.conversations.set(conversation.id, conversation);
          });
          console.log(`Loaded ${response.conversations.length} conversations`);
          
          // Load messages for each conversation
          Promise.all(response.conversations.map(conversation => 
            this._loadMessagesForConversation(conversation.id)
          )).then(() => resolve()).catch(err => reject(err));
        } else {
          console.error('Failed to load conversations:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Load messages for a specific conversation
   * @param {string} conversationId Conversation ID
   * @returns {Promise<void>}
   */
  async _loadMessagesForConversation(conversationId) {
    return new Promise((resolve, reject) => {
      this.socket.emit('get_messages', { 
        userID: this.userID,
        conversationId
      }, async (response) => {
        if (response.success) {
          const messages = [];
          
          // Decrypt all messages
          for (const encryptedMessage of response.messages) {
            try {
              const decryptedMessage = await this._decryptMessage(encryptedMessage);
              messages.push(decryptedMessage);
            } catch (error) {
              console.error(`Failed to decrypt message ${encryptedMessage.id}:`, error);
            }
          }
          
          // Store messages
          this.messages.set(conversationId, messages);
          console.log(`Loaded ${messages.length} messages for conversation ${conversationId}`);
          resolve();
        } else {
          console.error('Failed to load messages:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Handle conversation update
   * @param {Object} data Conversation update data
   */
  _handleConversationUpdate(data) {
    const { conversation, action } = data;
    
    switch (action) {
      case 'new':
      case 'update':
        this.conversations.set(conversation.id, conversation);
        console.log(`Conversation ${conversation.id} ${action === 'new' ? 'created' : 'updated'}`);
        break;
      case 'delete':
        this.conversations.delete(conversation.id);
        this.messages.delete(conversation.id);
        console.log(`Conversation ${conversation.id} deleted`);
        break;
      default:
        console.warn(`Unknown conversation action: ${action}`);
    }
    
    // Trigger conversation update listeners
    this._triggerConversationListeners(conversation, action);
  }
  
  /**
   * Create a new conversation
   * @param {string|string[]} recipients Recipient user ID or array of user IDs
   * @param {string} type Conversation type ('direct', 'group', 'team', 'trade')
   * @param {Object} metadata Additional conversation metadata
   * @returns {Promise<string>} Conversation ID
   */
  async createConversation(recipients, type = 'direct', metadata = {}) {
    return new Promise((resolve, reject) => {
      const recipientList = Array.isArray(recipients) ? recipients : [recipients];
      
      // Ensure all participants' public keys are available
      for (const recipient of recipientList) {
        if (!this.publicKeys.has(recipient)) {
          this._requestPublicKey(recipient);
        }
      }
      
      const conversation = {
        id: uuid.v4(),
        type,
        creator: this.userID,
        participants: [this.userID, ...recipientList],
        createdAt: new Date().toISOString(),
        lastActivity: new Date().toISOString(),
        metadata
      };
      
      this.socket.emit('create_conversation', conversation, (response) => {
        if (response.success) {
          this.conversations.set(conversation.id, conversation);
          this.messages.set(conversation.id, []);
          console.log(`Created conversation ${conversation.id}`);
          resolve(conversation.id);
        } else {
          console.error('Failed to create conversation:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Send message to conversation
   * @param {string} conversationId Conversation ID
   * @param {string} content Message content
   * @param {string} type Message type ('text', 'image', 'trade', etc.)
   * @param {Object} metadata Additional message metadata
   * @returns {Promise<Object>} Sent message
   */
  async sendMessage(conversationId, content, type = 'text', metadata = {}) {
    // Check if conversation exists
    const conversation = this.conversations.get(conversationId);
    if (!conversation) {
      throw new Error(`Conversation ${conversationId} not found`);
    }
    
    // Create message object
    const message = {
      id: uuid.v4(),
      conversationId,
      sender: this.userID,
      content,
      type,
      metadata,
      timestamp: new Date().toISOString(),
      status: 'sending'
    };
    
    // Store message locally first
    this._storeMessage(message);
    
    // Encrypt and send message to each participant
    try {
      const encryptedMessages = await Promise.all(
        conversation.participants
          .filter(participant => participant !== this.userID)
          .map(recipient => this._encryptMessageForRecipient(message, recipient))
      );
      
      return new Promise((resolve, reject) => {
        this.socket.emit('send_messages', { messages: encryptedMessages }, (response) => {
          if (response.success) {
            // Update message status
            message.status = 'sent';
            this._updateMessageStatus(message);
            console.log(`Message ${message.id} sent to conversation ${conversationId}`);
            resolve(message);
          } else {
            message.status = 'failed';
            this._updateMessageStatus(message);
            console.error('Failed to send message:', response.error);
            reject(new Error(response.error));
          }
        });
      });
    } catch (error) {
      message.status = 'failed';
      this._updateMessageStatus(message);
      console.error('Error encrypting message:', error);
      throw error;
    }
  }
  
  /**
   * Encrypt message for a specific recipient
   * @param {Object} message Message object
   * @param {string} recipientId Recipient user ID
   * @returns {Promise<Object>} Encrypted message
   */
  async _encryptMessageForRecipient(message, recipientId) {
    // Get recipient's public key
    const recipientPublicKey = this.publicKeys.get(recipientId);
    
    if (!recipientPublicKey) {
      throw new Error(`Public key not found for user ${recipientId}`);
    }
    
    // Create nonce (unique for each message)
    const nonce = this._generateNonce();
    
    // Stringify and encode the message content
    const messageData = JSON.stringify(message);
    const messageBytes = naclUtil.decodeUTF8(messageData);
    
    // Encrypt the message
    const encryptedBytes = nacl.box(
      messageBytes,
      nonce,
      recipientPublicKey,
      this.keyPair.secretKey
    );
    
    // Encode encrypted data and nonce to base64
    const encryptedBase64 = naclUtil.encodeBase64(encryptedBytes);
    const nonceBase64 = naclUtil.encodeBase64(nonce);
    
    // Return encrypted message
    return {
      id: message.id,
      conversationId: message.conversationId,
      sender: this.userID,
      recipient: recipientId,
      encryptedData: encryptedBase64,
      nonce: nonceBase64,
      timestamp: message.timestamp
    };
  }
  
  /**
   * Decrypt received message
   * @param {Object} encryptedMessage Encrypted message
   * @returns {Promise<Object>} Decrypted message
   */
  async _decryptMessage(encryptedMessage) {
    // Get sender's public key
    const senderPublicKey = this.publicKeys.get(encryptedMessage.sender);
    
    if (!senderPublicKey) {
      // Request public key from server
      await this._requestPublicKey(encryptedMessage.sender);
      
      // Check again
      if (!this.publicKeys.has(encryptedMessage.sender)) {
        throw new Error(`Public key not found for user ${encryptedMessage.sender}`);
      }
    }
    
    // Decode base64 data
    const encryptedBytes = naclUtil.decodeBase64(encryptedMessage.encryptedData);
    const nonce = naclUtil.decodeBase64(encryptedMessage.nonce);
    
    // Decrypt the message
    const decryptedBytes = nacl.box.open(
      encryptedBytes,
      nonce,
      senderPublicKey,
      this.keyPair.secretKey
    );
    
    if (!decryptedBytes) {
      throw new Error('Failed to decrypt message. The message may be corrupted or intended for someone else.');
    }
    
    // Decode and parse the message content
    const decryptedData = naclUtil.encodeUTF8(decryptedBytes);
    const message = JSON.parse(decryptedData);
    
    // Add received status
    message.status = 'received';
    
    return message;
  }
  
  /**
   * Request public key from server
   * @param {string} userId User ID
   * @returns {Promise<Uint8Array>} Public key
   */
  async _requestPublicKey(userId) {
    return new Promise((resolve, reject) => {
      this.socket.emit('get_public_key', { userID: userId }, (response) => {
        if (response.success) {
          const publicKey = naclUtil.decodeBase64(response.publicKey);
          this.publicKeys.set(userId, publicKey);
          console.log(`Retrieved public key for user ${userId}`);
          resolve(publicKey);
        } else {
          console.error('Failed to get public key:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Generate a unique nonce for message encryption
   * @returns {Uint8Array} 24-byte nonce
   */
  _generateNonce() {
    // Use counter and timestamp for deterministic but unique nonce
    const counter = this.messageCounter++;
    const timestamp = Date.now();
    
    // Create a deterministic input for the nonce
    const input = `${this.userID}-${timestamp}-${counter}`;
    const hash = createHash('sha256').update(input).digest();
    
    // Return first 24 bytes as nonce
    return new Uint8Array(hash.buffer, 0, 24);
  }
  
  /**
   * Store message in local storage
   * @param {Object} message Message object
   */
  _storeMessage(message) {
    const conversationId = message.conversationId;
    
    // Initialize message array if needed
    if (!this.messages.has(conversationId)) {
      this.messages.set(conversationId, []);
    }
    
    // Add message to conversation
    const conversationMessages = this.messages.get(conversationId);
    
    // Check if message already exists (by ID)
    const existingIndex = conversationMessages.findIndex(m => m.id === message.id);
    
    if (existingIndex >= 0) {
      // Update existing message
      conversationMessages[existingIndex] = { ...conversationMessages[existingIndex], ...message };
    } else {
      // Add new message
      conversationMessages.push(message);
      
      // Sort messages by timestamp
      conversationMessages.sort((a, b) => 
        new Date(a.timestamp) - new Date(b.timestamp)
      );
    }
    
    // Update conversation last activity
    const conversation = this.conversations.get(conversationId);
    if (conversation) {
      conversation.lastActivity = message.timestamp;
    }
  }
  
  /**
   * Update message status
   * @param {Object} message Message with updated status
   */
  _updateMessageStatus(message) {
    const conversationId = message.conversationId;
    const conversationMessages = this.messages.get(conversationId);
    
    if (!conversationMessages) return;
    
    const existingIndex = conversationMessages.findIndex(m => m.id === message.id);
    
    if (existingIndex >= 0) {
      conversationMessages[existingIndex].status = message.status;
    }
  }
  
  /**
   * Add message listener
   * @param {Function} callback Callback function for new messages
   * @returns {string} Listener ID
   */
  addMessageListener(callback) {
    const listenerId = uuid.v4();
    this.listeners.set(listenerId, callback);
    return listenerId;
  }
  
  /**
   * Remove message listener
   * @param {string} listenerId Listener ID
   */
  removeMessageListener(listenerId) {
    this.listeners.delete(listenerId);
  }
  
  /**
   * Trigger message listeners
   * @param {Object} message New message
   */
  _triggerMessageListeners(message) {
    this.listeners.forEach(callback => {
      try {
        callback(message);
      } catch (error) {
        console.error('Error in message listener:', error);
      }
    });
  }
  
  /**
   * Add conversation listener
   * @param {Function} callback Callback function for conversation updates
   * @returns {string} Listener ID
   */
  addConversationListener(callback) {
    const listenerId = uuid.v4();
    // Use a separate map for conversation listeners
    if (!this.conversationListeners) {
      this.conversationListeners = new Map();
    }
    this.conversationListeners.set(listenerId, callback);
    return listenerId;
  }
  
  /**
   * Remove conversation listener
   * @param {string} listenerId Listener ID
   */
  removeConversationListener(listenerId) {
    if (this.conversationListeners) {
      this.conversationListeners.delete(listenerId);
    }
  }
  
  /**
   * Trigger conversation listeners
   * @param {Object} conversation Updated conversation
   * @param {string} action Action performed on conversation
   */
  _triggerConversationListeners(conversation, action) {
    if (!this.conversationListeners) return;
    
    this.conversationListeners.forEach(callback => {
      try {
        callback(conversation, action);
      } catch (error) {
        console.error('Error in conversation listener:', error);
      }
    });
  }
  
  /**
   * Get all conversations
   * @returns {Array} Array of conversations
   */
  getConversations() {
    return Array.from(this.conversations.values()).sort((a, b) => 
      new Date(b.lastActivity) - new Date(a.lastActivity)
    );
  }
  
  /**
   * Get conversation by ID
   * @param {string} conversationId Conversation ID
   * @returns {Object} Conversation
   */
  getConversation(conversationId) {
    return this.conversations.get(conversationId);
  }
  
  /**
   * Get messages for conversation
   * @param {string} conversationId Conversation ID
   * @returns {Array} Array of messages
   */
  getMessages(conversationId) {
    return this.messages.get(conversationId) || [];
  }
  
  /**
   * Mark messages as read
   * @param {string} conversationId Conversation ID
   * @returns {Promise<boolean>} Success
   */
  async markAsRead(conversationId) {
    return new Promise((resolve, reject) => {
      this.socket.emit('mark_as_read', {
        userID: this.userID,
        conversationId
      }, (response) => {
        if (response.success) {
          // Update local message status
          const messages = this.messages.get(conversationId) || [];
          messages.forEach(message => {
            if (message.sender !== this.userID && message.status === 'received') {
              message.status = 'read';
            }
          });
          
          console.log(`Marked conversation ${conversationId} as read`);
          resolve(true);
        } else {
          console.error('Failed to mark as read:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Update conversation metadata
   * @param {string} conversationId Conversation ID
   * @param {Object} metadata New metadata
   * @returns {Promise<Object>} Updated conversation
   */
  async updateConversationMetadata(conversationId, metadata) {
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation) {
      throw new Error(`Conversation ${conversationId} not found`);
    }
    
    return new Promise((resolve, reject) => {
      const updatedConversation = {
        ...conversation,
        metadata: { ...conversation.metadata, ...metadata },
        lastActivity: new Date().toISOString()
      };
      
      this.socket.emit('update_conversation', updatedConversation, (response) => {
        if (response.success) {
          this.conversations.set(conversationId, updatedConversation);
          console.log(`Updated conversation ${conversationId} metadata`);
          resolve(updatedConversation);
        } else {
          console.error('Failed to update conversation:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Create a trade negotiation conversation
   * @param {string} recipientId Recipient user ID
   * @param {Object} initialOffer Initial trade offer
   * @returns {Promise<string>} Conversation ID
   */
  async createTradeNegotiation(recipientId, initialOffer) {
    const metadata = {
      tradeStatus: 'proposed',
      offer: initialOffer
    };
    
    const conversationId = await this.createConversation(
      recipientId,
      'trade',
      metadata
    );
    
    // Send initial trade message
    await this.sendMessage(
      conversationId,
      'Trade proposal',
      'trade_offer',
      { offer: initialOffer }
    );
    
    return conversationId;
  }
  
  /**
   * Update trade offer in a trade negotiation
   * @param {string} conversationId Conversation ID
   * @param {Object} newOffer New trade offer
   * @returns {Promise<Object>} Sent message
   */
  async updateTradeOffer(conversationId, newOffer) {
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation || conversation.type !== 'trade') {
      throw new Error('Invalid trade conversation');
    }
    
    // Update conversation metadata
    await this.updateConversationMetadata(conversationId, {
      tradeStatus: 'negotiating',
      offer: newOffer
    });
    
    // Send update message
    return await this.sendMessage(
      conversationId,
      'Updated trade offer',
      'trade_update',
      { offer: newOffer }
    );
  }
  
  /**
   * Accept trade offer
   * @param {string} conversationId Conversation ID
   * @returns {Promise<Object>} Sent message
   */
  async acceptTradeOffer(conversationId) {
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation || conversation.type !== 'trade') {
      throw new Error('Invalid trade conversation');
    }
    
    // Update conversation metadata
    await this.updateConversationMetadata(conversationId, {
      tradeStatus: 'accepted'
    });
    
    // Send acceptance message
    return await this.sendMessage(
      conversationId,
      'Trade accepted',
      'trade_accepted',
      { timestamp: new Date().toISOString() }
    );
  }
  
  /**
   * Reject trade offer
   * @param {string} conversationId Conversation ID
   * @param {string} reason Rejection reason
   * @returns {Promise<Object>} Sent message
   */
  async rejectTradeOffer(conversationId, reason = '') {
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation || conversation.type !== 'trade') {
      throw new Error('Invalid trade conversation');
    }
    
    // Update conversation metadata
    await this.updateConversationMetadata(conversationId, {
      tradeStatus: 'rejected'
    });
    
    // Send rejection message
    return await this.sendMessage(
      conversationId,
      'Trade rejected',
      'trade_rejected',
      { 
        reason,
        timestamp: new Date().toISOString()
      }
    );
  }
  
  /**
   * Create a team chat
   * @param {string} teamId Team ID
   * @param {string[]} teamMembers Team member user IDs
   * @returns {Promise<string>} Conversation ID
   */
  async createTeamChat(teamId, teamMembers) {
    const metadata = {
      teamId,
      isTeamChat: true
    };
    
    const conversationId = await this.createConversation(
      teamMembers,
      'team',
      metadata
    );
    
    // Send welcome message
    await this.sendMessage(
      conversationId,
      'Team chat created',
      'system',
      { event: 'team_chat_created' }
    );
    
    return conversationId;
  }
  
  /**
   * Add user to conversation
   * @param {string} conversationId Conversation ID
   * @param {string} userId User ID to add
   * @returns {Promise<Object>} Updated conversation
   */
  async addParticipant(conversationId, userId) {
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation) {
      throw new Error(`Conversation ${conversationId} not found`);
    }
    
    if (conversation.participants.includes(userId)) {
      return conversation; // User already in conversation
    }
    
    return new Promise((resolve, reject) => {
      const updatedConversation = {
        ...conversation,
        participants: [...conversation.participants, userId],
        lastActivity: new Date().toISOString()
      };
      
      this.socket.emit('update_conversation', updatedConversation, (response) => {
        if (response.success) {
          this.conversations.set(conversationId, updatedConversation);
          
          // Send system message about the new participant
          this.sendMessage(
            conversationId,
            `User ${userId} joined the conversation`,
            'system',
            { event: 'user_joined', userId }
          ).catch(err => console.error('Error sending system message:', err));
          
          console.log(`Added user ${userId} to conversation ${conversationId}`);
          resolve(updatedConversation);
        } else {
          console.error('Failed to add participant:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Remove user from conversation
   * @param {string} conversationId Conversation ID
   * @param {string} userId User ID to remove
   * @returns {Promise<Object>} Updated conversation
   */
  async removeParticipant(conversationId, userId) {
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation) {
      throw new Error(`Conversation ${conversationId} not found`);
    }
    
    if (!conversation.participants.includes(userId)) {
      return conversation; // User not in conversation
    }
    
    return new Promise((resolve, reject) => {
      const updatedConversation = {
        ...conversation,
        participants: conversation.participants.filter(id => id !== userId),
        lastActivity: new Date().toISOString()
      };
      
      this.socket.emit('update_conversation', updatedConversation, (response) => {
        if (response.success) {
          this.conversations.set(conversationId, updatedConversation);
          
          // Send system message about the participant leaving
          this.sendMessage(
            conversationId,
            `User ${userId} left the conversation`,
            'system',
            { event: 'user_left', userId }
          ).catch(err => console.error('Error sending system message:', err));
          
          console.log(`Removed user ${userId} from conversation ${conversationId}`);
          resolve(updatedConversation);
        } else {
          console.error('Failed to remove participant:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Delete conversation
   * @param {string} conversationId Conversation ID
   * @returns {Promise<boolean>} Success
   */
  async deleteConversation(conversationId) {
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation) {
      throw new Error(`Conversation ${conversationId} not found`);
    }
    
    return new Promise((resolve, reject) => {
      this.socket.emit('delete_conversation', {
        userID: this.userID,
        conversationId
      }, (response) => {
        if (response.success) {
          this.conversations.delete(conversationId);
          this.messages.delete(conversationId);
          console.log(`Deleted conversation ${conversationId}`);
          resolve(true);
        } else {
          console.error('Failed to delete conversation:', response.error);
          reject(new Error(response.error));
        }
      });
    });
  }
  
  /**
   * Clean up and disconnect
   */
  disconnect() {
    if (this.socket) {
      this.socket.disconnect();
      this.socket = null;
    }
    
    this.connected = false;
    console.log('Disconnected from messaging system');
  }
}

/**
 * Helper for client-side message encryption
 */
class MessageCrypto {
  /**
   * Generate a new key pair
   * @returns {Object} Object containing public and private keys
   */
  static generateKeyPair() {
    const keyPair = nacl.box.keyPair();
    return {
      publicKey: naclUtil.encodeBase64(keyPair.publicKey),
      privateKey: naclUtil.encodeBase64(keyPair.secretKey)
    };
  }
  
  /**
   * Encrypt message for direct sending (without server)
   * @param {string} message Message to encrypt
   * @param {string} recipientPublicKey Recipient's public key (base64)
   * @param {string} senderPrivateKey Sender's private key (base64)
   * @returns {Object} Encrypted message data
   */
  static encryptMessage(message, recipientPublicKey, senderPrivateKey) {
    // Decode keys from base64
    const publicKeyBytes = naclUtil.decodeBase64(recipientPublicKey);
    const privateKeyBytes = naclUtil.decodeBase64(senderPrivateKey);
    
    // Create nonce
    const nonce = nacl.randomBytes(24);
    
    // Encrypt the message
    const messageBytes = naclUtil.decodeUTF8(message);
    const encryptedBytes = nacl.box(
      messageBytes,
      nonce,
      publicKeyBytes,
      privateKeyBytes
    );
    
    // Return encrypted data and nonce
    return {
      encryptedData: naclUtil.encodeBase64(encryptedBytes),
      nonce: naclUtil.encodeBase64(nonce)
    };
  }
  
  /**
   * Decrypt message (direct decryption)
   * @param {string} encryptedData Encrypted data (base64)
   * @param {string} nonce Nonce used for encryption (base64)
   * @param {string} senderPublicKey Sender's public key (base64)
   * @param {string} recipientPrivateKey Recipient's private key (base64)
   * @returns {string} Decrypted message
   */
  static decryptMessage(encryptedData, nonce, senderPublicKey, recipientPrivateKey) {
    // Decode from base64
    const encryptedBytes = naclUtil.decodeBase64(encryptedData);
    const nonceBytes = naclUtil.decodeBase64(nonce);
    const publicKeyBytes = naclUtil.decodeBase64(senderPublicKey);
    const privateKeyBytes = naclUtil.decodeBase64(recipientPrivateKey);
    
    // Decrypt the message
    const decryptedBytes = nacl.box.open(
      encryptedBytes,
      nonceBytes,
      publicKeyBytes,
      privateKeyBytes
    );
    
    if (!decryptedBytes) {
      throw new Error('Failed to decrypt message');
    }
    
    // Return decrypted message
    return naclUtil.encodeUTF8(decryptedBytes);
  }
}

/**
 * Messaging server implementation
 * This would be implemented as a separate backend service
 */
class MessagingServer {
  constructor() {
    this.io = null;
    this.publicKeys = new Map(); // User ID to public key
    this.conversations = new Map(); // Conversation ID to conversation data
    this.messages = new Map(); // Conversation ID to array of messages
    this.userConversations = new Map(); // User ID to set of conversation IDs
    this.connectedUsers = new Map(); // User ID to socket ID
  }
  
  /**
   * Initialize the messaging server
   * @param {Object} server HTTP server to attach socket.io to
   */
  initialize(server) {
    this.io = require('socket.io')(server, {
      cors: {
        origin: '*', // In production, restrict to your domains
        methods: ['GET', 'POST']
      }
    });
    
    this._setupSocketHandlers();
    
    console.log('Messaging server initialized');
  }
  
  /**
   * Set up socket handlers
   */
  _setupSocketHandlers() {
    this.io.on('connection', (socket) => {
      const userID = socket.handshake.query.userID;
      
      if (!userID) {
        console.error('Connection attempt without userID');
        socket.disconnect();
        return;
      }
      
      console.log(`User ${userID} connected`);
      
      // Store connection
      this.connectedUsers.set(userID, socket.id);
      
      // Handle public key registration
      socket.on('register_public_key', (data, callback) => {
        this._handleRegisterPublicKey(data, callback);
      });
      
      // Handle get public key
      socket.on('get_public_key', (data, callback) => {
        this._handleGetPublicKey(data, callback);
      });
      
      // Handle create conversation
      socket.on('create_conversation', (data, callback) => {
        this._handleCreateConversation(data, callback);
      });
      
      // Handle update conversation
      socket.on('update_conversation', (data, callback) => {
        this._handleUpdateConversation(data, callback);
      });
      
      // Handle delete conversation
      socket.on('delete_conversation', (data, callback) => {
        this._handleDeleteConversation(data, callback);
      });
      
      // Handle get conversations
      socket.on('get_conversations', (data, callback) => {
        this._handleGetConversations(data, callback);
      });
      
      // Handle send messages
      socket.on('send_messages', (data, callback) => {
        this._handleSendMessages(data, callback);
      });
      
      // Handle get messages
      socket.on('get_messages', (data, callback) => {
        this._handleGetMessages(data, callback);
      });
      
      // Handle mark as read
      socket.on('mark_as_read', (data, callback) => {
        this._handleMarkAsRead(data, callback);
      });
      
      // Handle disconnection
      socket.on('disconnect', () => {
        console.log(`User ${userID} disconnected`);
        this.connectedUsers.delete(userID);
      });
    });
  }
  
  /**
   * Handle public key registration
   * @param {Object} data Registration data
   * @param {Function} callback Callback function
   */
  _handleRegisterPublicKey(data, callback) {
    const { userID, publicKey } = data;
    
    if (!userID || !publicKey) {
      callback({ success: false, error: 'Missing userID or publicKey' });
      return;
    }
    
    // Store public key
    this.publicKeys.set(userID, publicKey);
    console.log(`Registered public key for user ${userID}`);
    
    // Notify other users that might be interested
    this._notifyPublicKeyUpdate(userID, publicKey);
    
    callback({ success: true });
  }
  
  /**
   * Notify users of public key update
   * @param {string} userID User ID
   * @param {string} publicKey Public key
   */
  _notifyPublicKeyUpdate(userID, publicKey) {
    // Find users who have conversations with this user
    const interestedUsers = new Set();
    
    for (const [conversationId, conversation] of this.conversations.entries()) {
      if (conversation.participants.includes(userID)) {
        for (const participant of conversation.participants) {
          if (participant !== userID) {
            interestedUsers.add(participant);
          }
        }
      }
    }
    
    // Notify interested users
    for (const interestedUser of interestedUsers) {
      const socketId = this.connectedUsers.get(interestedUser);
      if (socketId) {
        this.io.to(socketId).emit('public_key_update', {
          userID,
          publicKey
        });
      }
    }
  }
  
  /**
   * Handle get public key request
   * @param {Object} data Request data
   * @param {Function} callback Callback function
   */
  _handleGetPublicKey(data, callback) {
    const { userID } = data;
    
    if (!userID) {
      callback({ success: false, error: 'Missing userID' });
      return;
    }
    
    const publicKey = this.publicKeys.get(userID);
    
    if (!publicKey) {
      callback({ success: false, error: 'Public key not found' });
      return;
    }
    
    callback({ success: true, publicKey });
  }
  
  /**
   * Handle create conversation
   * @param {Object} conversation Conversation data
   * @param {Function} callback Callback function
   */
  _handleCreateConversation(conversation, callback) {
    if (!conversation.id || !conversation.participants || conversation.participants.length === 0) {
      callback({ success: false, error: 'Invalid conversation data' });
      return;
    }
    
    // Store conversation
    this.conversations.set(conversation.id, conversation);
    
    // Initialize message array
    this.messages.set(conversation.id, []);
    
    // Update user conversations
    for (const participant of conversation.participants) {
      if (!this.userConversations.has(participant)) {
        this.userConversations.set(participant, new Set());
      }
      this.userConversations.get(participant).add(conversation.id);
    }
    
    // Notify participants
    this._notifyConversationUpdate(conversation, 'new');
    
    callback({ success: true, conversation });
  }
  
  /**
   * Handle update conversation
   * @param {Object} conversation Updated conversation data
   * @param {Function} callback Callback function
   */
  _handleUpdateConversation(conversation, callback) {
    const existingConversation = this.conversations.get(conversation.id);
    
    if (!existingConversation) {
      callback({ success: false, error: 'Conversation not found' });
      return;
    }
    
    // Update conversation
    this.conversations.set(conversation.id, conversation);
    
    // Update user conversations if participants changed
    const oldParticipants = new Set(existingConversation.participants);
    const newParticipants = new Set(conversation.participants);
    
    // Remove users no longer in conversation
    for (const participant of oldParticipants) {
      if (!newParticipants.has(participant)) {
        const userConversations = this.userConversations.get(participant);
        if (userConversations) {
          userConversations.delete(conversation.id);
        }
      }
    }
    
    // Add new users to conversation
    for (const participant of newParticipants) {
      if (!oldParticipants.has(participant)) {
        if (!this.userConversations.has(participant)) {
          this.userConversations.set(participant, new Set());
        }
        this.userConversations.get(participant).add(conversation.id);
      }
    }
    
    // Notify participants
    this._notifyConversationUpdate(conversation, 'update');
    
    callback({ success: true, conversation });
  }
  
  /**
   * Notify participants of conversation update
   * @param {Object} conversation Conversation data
   * @param {string} action Action performed
   */
  _notifyConversationUpdate(conversation, action) {
    for (const participant of conversation.participants) {
      const socketId = this.connectedUsers.get(participant);
      if (socketId) {
        this.io.to(socketId).emit('conversation_update', {
          conversation,
          action
        });
      }
    }
  }
  
  /**
   * Handle delete conversation
   * @param {Object} data Request data
   * @param {Function} callback Callback function
   */
  _handleDeleteConversation(data, callback) {
    const { conversationId, userID } = data;
    
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation) {
      callback({ success: false, error: 'Conversation not found' });
      return;
    }
    
    // Check if user is in conversation
    if (!conversation.participants.includes(userID)) {
      callback({ success: false, error: 'User not in conversation' });
      return;
    }
    
    // Notify participants before deletion
    this._notifyConversationUpdate(conversation, 'delete');
    
    // Delete conversation
    this.conversations.delete(conversationId);
    this.messages.delete(conversationId);
    
    // Update user conversations
    for (const participant of conversation.participants) {
      const userConversations = this.userConversations.get(participant);
      if (userConversations) {
        userConversations.delete(conversationId);
      }
    }
    
    callback({ success: true });
  }
  
  /**
   * Handle get conversations
   * @param {Object} data Request data
   * @param {Function} callback Callback function
   */
  _handleGetConversations(data, callback) {
    const { userID } = data;
    
    if (!userID) {
      callback({ success: false, error: 'Missing userID' });
      return;
    }
    
    const userConversationIds = this.userConversations.get(userID) || new Set();
    const conversations = [];
    
    for (const conversationId of userConversationIds) {
      const conversation = this.conversations.get(conversationId);
      if (conversation) {
        conversations.push(conversation);
      }
    }
    
    callback({ success: true, conversations });
  }
  
  /**
   * Handle send messages
   * @param {Object} data Message data
   * @param {Function} callback Callback function
   */
  _handleSendMessages(data, callback) {
    const { messages } = data;
    
    if (!messages || messages.length === 0) {
      callback({ success: false, error: 'No messages to send' });
      return;
    }
    
    // Process each message
    for (const message of messages) {
      const { conversationId, recipient } = message;
      
      // Store message
      const conversationMessages = this.messages.get(conversationId) || [];
      conversationMessages.push(message);
      this.messages.set(conversationId, conversationMessages);
      
      // Send to recipient if online
      const socketId = this.connectedUsers.get(recipient);
      if (socketId) {
        this.io.to(socketId).emit('encrypted_message', message);
      }
    }
    
    callback({ success: true });
  }
  
  /**
   * Handle get messages
   * @param {Object} data Request data
   * @param {Function} callback Callback function
   */
  _handleGetMessages(data, callback) {
    const { userID, conversationId } = data;
    
    if (!userID || !conversationId) {
      callback({ success: false, error: 'Missing userID or conversationId' });
      return;
    }
    
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation) {
      callback({ success: false, error: 'Conversation not found' });
      return;
    }
    
    // Check if user is in conversation
    if (!conversation.participants.includes(userID)) {
      callback({ success: false, error: 'User not in conversation' });
      return;
    }
    
    // Get messages for conversation
    const messages = this.messages.get(conversationId) || [];
    
    // Filter to messages intended for this user
    const userMessages = messages.filter(message => 
      message.recipient === userID ||
      message.sender === userID
    );
    
    callback({ success: true, messages: userMessages });
  }
  
  /**
   * Handle mark as read
   * @param {Object} data Request data
   * @param {Function} callback Callback function
   */
  _handleMarkAsRead(data, callback) {
    const { userID, conversationId } = data;
    
    if (!userID || !conversationId) {
      callback({ success: false, error: 'Missing userID or conversationId' });
      return;
    }
    
    const conversation = this.conversations.get(conversationId);
    
    if (!conversation) {
      callback({ success: false, error: 'Conversation not found' });
      return;
    }
    
    // Check if user is in conversation
    if (!conversation.participants.includes(userID)) {
      callback({ success: false, error: 'User not in conversation' });
      return;
    }
    
    // Update conversation last read (in a real implementation)
    // ...
    
    callback({ success: true });
  }
}
