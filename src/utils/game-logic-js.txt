import { getMovementPattern } from './movementPatterns';

// Constants
export const BOARD_COLUMNS = 20;
export const BOARD_ROWS = 10;
export const MOVES_PER_TURN = 3;
export const MATCH_LENGTH_MINUTES = 90; // Simulated match time

// Game states
export const GAME_STATES = {
  SETUP: 'setup',
  PLAYING: 'playing',
  HALF_TIME: 'half_time',
  FULL_TIME: 'full_time',
  GOAL_SCORED: 'goal_scored'
};

// Team sides
export const TEAMS = {
  HOME: 'home',
  AWAY: 'away'
};

/**
 * Create a clean game board
 * @returns {Array} 2D array representing the empty game board
 */
export function createEmptyBoard() {
  return Array(BOARD_COLUMNS).fill().map(() => Array(BOARD_ROWS).fill(null));
}

/**
 * Initialize a new game with players in starting positions
 * @param {Object} homeTeam Team data for home team
 * @param {Object} awayTeam Team data for away team
 * @returns {Object} Initial game state
 */
export function initializeGame(homeTeam, awayTeam) {
  const board = createEmptyBoard();
  const players = [...generateTeamPlayers(homeTeam, TEAMS.HOME), ...generateTeamPlayers(awayTeam, TEAMS.AWAY)];
  
  // Place players on the board
  players.forEach(player => {
    if (board[player.col][player.row]) {
      console.warn(`Position conflict at ${player.col},${player.row}. Multiple players at same spot.`);
    }
    board[player.col][player.row] = player;
  });
  
  // Initial ball position (with the home team striker)
  const initialBallHolder = players.find(p => p.team === TEAMS.HOME && p.position === 'ST');
  const ball = {
    col: initialBallHolder ? initialBallHolder.col : 9, 
    row: initialBallHolder ? initialBallHolder.row : 5,
    possessionPlayerId: initialBallHolder ? initialBallHolder.id : null
  };
  
  if (initialBallHolder) {
    initialBallHolder.hasBall = true;
  }
  
  return {
    board,
    players,
    ball,
    score: { home: 0, away: 0 },
    currentTeam: TEAMS.HOME,
    movesRemaining: MOVES_PER_TURN,
    gameState: GAME_STATES.PLAYING,
    turnNumber: 1,
    matchMinute: 0,
    gameLog: [
      { type: 'info', text: 'Match started!' },
      { type: 'info', text: 'Home team to play first' }
    ],
    fouls: [],
    injuries: [],
    possession: { home: 0, away: 0 }
  };
}

/**
 * Create players for a team based on team data
 * @param {Object} team Team data
 * @param {String} side Which side (home/away)
 * @returns {Array} Array of player objects
 */
function generateTeamPlayers(team, side) {
  const isHome = side === TEAMS.HOME;
  const defaultFormation = isHome 
    ? { formation: '4-3-3', kickoffPositions: getKickoffPositions(TEAMS.HOME) }
    : { formation: '4-3-3', kickoffPositions: getKickoffPositions(TEAMS.AWAY) };
  
  const { formation, kickoffPositions } = team.formation || defaultFormation;
  
  return team.players.map(player => {
    const position = kickoffPositions.find(pos => pos.position === player.position);
    
    if (!position) {
      console.warn(`No kickoff position found for ${player.name} (${player.position})`);
      return null;
    }
    
    return {
      id: player.id,
      name: player.name,
      team: side,
      position: player.position,
      number: player.number,
      col: position.col,
      row: position.row,
      attributes: player.attributes || {
        pace: 70,
        shooting: 70,
        passing: 70,
        dribbling: 70,
        defending: 70,
        physical: 70
      },
      hasBall: false,
      yellowCard: false,
      redCard: false,
      injury: null,
      onField: true,
      movementReduction: 0
    };
  }).filter(Boolean); // Remove any null entries
}

/**
 * Get default kickoff positions for a team
 * @param {String} team The team (home/away)
 * @returns {Array} Array of position objects
 */
function getKickoffPositions(team) {
  if (team === TEAMS.HOME) {
    return [
      { position: 'GK', col: 1, row: 5 },
      { position: 'LB', col: 3, row: 2 },
      { position: 'CB', col: 3, row: 4 },
      { position: 'CB', col: 3, row: 6 },
      { position: 'RB', col: 3, row: 8 },
      { position: 'CDM', col: 5, row: 5 },
      { position: 'CM', col: 7, row: 4 },
      { position: 'CM', col: 7, row: 6 },
      { position: 'LW', col: 9, row: 2 },
      { position: 'ST', col: 9, row: 5, hasBall: true }, // Striker starts with the ball
      { position: 'RW', col: 9, row: 8 }
    ];
  } else {
    return [
      { position: 'GK', col: 18, row: 5 },
      { position: 'LB', col: 16, row: 2 },
      { position: 'CB', col: 16, row: 4 },
      { position: 'CB', col: 16, row: 6 },
      { position: 'RB', col: 16, row: 8 },
      { position: 'CDM', col: 14, row: 5 },
      { position: 'CM', col: 12, row: 4 },
      { position: 'CM', col: 12, row: 6 },
      { position: 'LW', col: 10, row: 2 },
      { position: 'ST', col: 10, row: 5 },
      { position: 'RW', col: 10, row: 8 }
    ];
  }
}

/**
 * Generate possible moves for a player based on their position and movement pattern
 * @param {Object} player Player object
 * @param {Array} board Current game board state
 * @returns {Array} Array of valid move coordinates
 */
export function getPossibleMoves(player, board) {
  if (!player || !player.onField || player.redCard) {
    return [];
  }
  
  // Get the base movement pattern for this position
  const movements = getMovementPattern(player.position);
  
  // Apply any movement reductions from injuries
  const actualRange = Math.max(1, movements.range - (player.movementReduction || 0));
  
  const possibleMoves = [];
  const { col, row } = player;
  
  // Apply the movement pattern with the actual range
  movements.directions.forEach(dir => {
    const { dx, dy } = dir;
    
    for (let i = 1; i <= actualRange; i++) {
      const newCol = col + (dx * i);
      const newRow = row + (dy * i);
      
      // Check if the move is within board boundaries
      if (isValidPosition(newCol, newRow)) {
        possibleMoves.push({ col: newCol, row: newRow });
      }
    }
  });
  
  // Add any special moves for this position
  if (movements.specialMoves) {
    movements.specialMoves.forEach(move => {
      const newCol = col + move.dx;
      const newRow = row + move.dy;
      
      if (isValidPosition(newCol, newRow)) {
        possibleMoves.push({ col: newCol, row: newRow });
      }
    });
  }
  
  return possibleMoves;
}

/**
 * Check if a position is valid on the board
 * @param {Number} col Column index
 * @param {Number} row Row index
 * @returns {Boolean} True if position is valid
 */
export function isValidPosition(col, row) {
  return col >= 0 && col < BOARD_COLUMNS && row >= 0 && row < BOARD_ROWS;
}

/**
 * Move a player to a new position
 * @param {Object} gameState Current game state
 * @param {String} playerId ID of the player to move
 * @param {Number} newCol New column position
 * @param {Number} newRow New row position
 * @returns {Object} Updated game state
 */
export function movePlayer(gameState, playerId, newCol, newRow) {
  const { board, players, ball, currentTeam, movesRemaining } = gameState;
  
  // Find the player
  const playerIndex = players.findIndex(p => p.id === playerId);
  if (playerIndex === -1) return gameState;
  
  const player = players[playerIndex];
  
  // Validate that it's this player's team's turn
  if (player.team !== currentTeam) return gameState;
  
  // Validate that the player can move to the target position
  const possibleMoves = getPossibleMoves(player, board);
  const isValidMove = possibleMoves.some(move => move.col === newCol && move.row === newRow);
  
  if (!isValidMove) return gameState;
  
  // Create new board and player objects to preserve immutability
  const newBoard = board.map(col => [...col]);
  const newPlayers = [...players];
  const newPlayer = { ...player };
  newPlayers[playerIndex] = newPlayer;
  
  // Update ball position if the player has the ball
  const newBall = { ...ball };
  const hasBall = player.hasBall;
  
  // Clear the old position on the board
  newBoard[player.col][player.row] = null;
  
  // Update player position
  newPlayer.col = newCol;
  newPlayer.row = newRow;
  
  // Place player in new position on board
  newBoard[newCol][newRow] = newPlayer;
  
  // Move the ball with the player if they have possession
  if (hasBall) {
    newBall.col = newCol;
    newBall.row = newRow;
    newBall.possessionPlayerId = playerId;
  }
  
  // Check for possession changes based on players in the square
  const updatedGameState = checkPossessionChange({
    ...gameState,
    board: newBoard,
    players: newPlayers,
    ball: newBall,
    movesRemaining: movesRemaining - 1
  }, newCol, newRow);
  
  // Check for fouls when moving to a square with opponents
  const gameStateWithFouls = checkForFouls(updatedGameState, newCol, newRow);
  
  // Log the move
  const moveLog = {
    type: 'move',
    text: hasBall 
      ? `${player.name} moved with the ball to ${newCol},${newRow}` 
      : `${player.name} moved to ${newCol},${newRow}`
  };
  
  // End turn if no moves remaining
  let finalGameState = {
    ...gameStateWithFouls,
    gameLog: [...gameStateWithFouls.gameLog, moveLog]
  };
  
  if (finalGameState.movesRemaining <= 0) {
    finalGameState = endTurn(finalGameState);
  }
  
  return finalGameState;
}

/**
 * Check for possession changes when multiple players are on the same square
 * @param {Object} gameState Current game state
 * @param {Number} col Column to check
 * @param {Number} row Row to check
 * @returns {Object} Updated game state
 */
function checkPossessionChange(gameState, col, row) {
  const { players, ball } = gameState;
  
  // If the ball is at this location
  if (ball.col === col && ball.row === row) {
    // Find all players at this location
    const playersAtLocation = players.filter(p => p.onField && p.col === col && p.row === row);
    const homeCount = playersAtLocation.filter(p => p.team === TEAMS.HOME).length;
    const awayCount = playersAtLocation.filter(p => p.team === TEAMS.AWAY).length;
    
    // The team with more players gets possession
    if (homeCount > awayCount) {
      // Give ball to a home player
      return updateBallPossession(gameState, playersAtLocation.find(p => p.team === TEAMS.HOME).id);
    } else if (awayCount > homeCount) {
      // Give ball to an away player
      return updateBallPossession(gameState, playersAtLocation.find(p => p.team === TEAMS.AWAY).id);
    }
    // If equal, ball remains where it is but no player has possession
  }
  
  return gameState;
}

/**
 * Update ball possession to a specific player
 * @param {Object} gameState Current game state
 * @param {String} playerId ID of the player to give the ball to
 * @returns {Object} Updated game state
 */
export function updateBallPossession(gameState, playerId) {
  const { players, ball } = gameState;
  
  // Find the player
  const playerIndex = players.findIndex(p => p.id === playerId);
  if (playerIndex === -1) return gameState;
  
  const player = players[playerIndex];
  
  // Create new arrays to preserve immutability
  const newPlayers = players.map(p => ({
    ...p,
    hasBall: p.id === playerId
  }));
  
  const newBall = {
    ...ball,
    col: player.col,
    row: player.row,
    possessionPlayerId: playerId
  };
  
  // Update possession statistics
  const possessionUpdate = { ...gameState.possession };
  possessionUpdate[player.team]++;
  
  // Log the possession change
  const possessionLog = {
    type: 'action',
    text: `${player.name} now has possession of the ball`
  };
  
  return {
    ...gameState,
    players: newPlayers,
    ball: newBall,
    possession: possessionUpdate,
    gameLog: [...gameState.gameLog, possessionLog]
  };
}

/**
 * Check for fouls when multiple players from opposing teams are on the same square
 * @param {Object} gameState Current game state
 * @param {Number} col Column to check
 * @param {Number} row Row to check
 * @returns {Object} Updated game state
 */
function checkForFouls(gameState, col, row) {
  // In a real implementation, this would have complex logic to determine
  // if a foul occurs based on player attributes, proximity to goal, etc.
  // For this example, we'll use a simplified random chance
  
  const { players, currentTeam } = gameState;
  
  // Find all players at this location
  const playersAtLocation = players.filter(p => p.onField && p.col === col && p.row === row);
  const homeCount = playersAtLocation.filter(p => p.team === TEAMS.HOME).length;
  const awayCount = playersAtLocation.filter(p => p.team === TEAMS.AWAY).length;
  
  // Only check for fouls if there are players from both teams
  if (homeCount > 0 && awayCount > 0) {
    // Determine which team could commit the foul (the team not in possession)
    const potentialFoulingTeam = currentTeam === TEAMS.HOME ? TEAMS.AWAY : TEAMS.HOME;
    
    // Simple random chance of a foul
    const foulChance = 0.15; // 15% chance
    
    if (Math.random() < foulChance) {
      // A foul has occurred!
      return handleFoul(gameState, col, row, potentialFoulingTeam);
    }
  }
  
  return gameState;
}

/**
 * Handle a foul occurrence
 * @param {Object} gameState Current game state
 * @param {Number} col Column where the foul occurred
 * @param {Number} row Row where the foul occurred
 * @param {String} foulingTeam Team that committed the foul
 * @returns {Object} Updated game state
 */
function handleFoul(gameState, col, row, foulingTeam) {
  // Determine foul severity
  const severityRoll = Math.random();
  let foulType;
  
  // Check if the foul is in a scoring position
  const isScoringPosition = (foulingTeam === TEAMS.HOME && col >= 15) || 
                           (foulingTeam === TEAMS.AWAY && col <= 4);
  
  // Determine foul type and card
  if (severityRoll < 0.6) {
    // Soft foul - 60% chance
    foulType = isScoringPosition ? 'yellow' : 'warning';
  } else if (severityRoll < 0.9) {
    // Hard foul - 30% chance
    foulType = isScoringPosition ? 'red' : 'yellow';
  } else {
    // Injury foul - 10% chance
    foulType = 'red';
  }
  
  // Process the card
  let updatedGameState = gameState;
  
  if (foulType === 'warning') {
    updatedGameState = issueWarning(gameState, foulingTeam, col, row);
  } else if (foulType === 'yellow') {
    updatedGameState = issueYellowCard(gameState, foulingTeam, col, row);
  } else if (foulType === 'red') {
    updatedGameState = issueRedCard(gameState, foulingTeam, col, row);
    
    // Red cards might also cause injuries
    if (severityRoll >= 0.9) {
      updatedGameState = causeInjury(updatedGameState, foulingTeam === TEAMS.HOME ? TEAMS.AWAY : TEAMS.HOME, col, row);
    }
  }
  
  return updatedGameState;
}

/**
 * Issue a warning to a random player on the specified team
 * @param {Object} gameState Current game state
 * @param {String} team Team to issue warning to
 * @param {Number} col Column location of the foul
 * @param {Number} row Row location of the foul
 * @returns {Object} Updated game state
 */
function issueWarning(gameState, team, col, row) {
  // Find players from the team in the vicinity
  const playersNearby = gameState.players.filter(
    p => p.team === team && p.onField && 
    Math.abs(p.col - col) <= 1 && Math.abs(p.row - row) <= 1
  );
  
  if (playersNearby.length === 0) return gameState;
  
  // Pick a random player to warn
  const randomIndex = Math.floor(Math.random() * playersNearby.length);
  const player = playersNearby[randomIndex];
  
  // Log the warning
  const warningLog = {
    type: 'warning',
    text: `Warning issued to ${player.name} for a foul`
  };
  
  // Add to fouls list
  const newFouls = [...gameState.fouls, {
    type: 'warning',
    player: player.id,
    team,
    minute: gameState.matchMinute,
    location: { col, row }
  }];
  
  return {
    ...gameState,
    gameLog: [...gameState.gameLog, warningLog],
    fouls: newFouls
  };
}

/**
 * Issue a yellow card to a random player on the specified team
 * @param {Object} gameState Current game state
 * @param {String} team Team to issue card to
 * @param {Number} col Column location of the foul
 * @param {Number} row Row location of the foul
 * @returns {Object} Updated game state
 */
function issueYellowCard(gameState, team, col, row) {
  // Find players from the team in the vicinity
  const playersNearby = gameState.players.filter(
    p => p.team === team && p.onField && !p.redCard &&
    Math.abs(p.col - col) <= 1 && Math.abs(p.row - row) <= 1
  );
  
  if (playersNearby.length === 0) return gameState;
  
  // Pick a random player to card
  const randomIndex = Math.floor(Math.random() * playersNearby.length);
  const player = playersNearby[randomIndex];
  
  // Create new players array
  const newPlayers = gameState.players.map(p => {
    if (p.id !== player.id) return p;
    
    // Check if this is a second yellow card
    if (p.yellowCard) {
      // Second yellow = red card
      return {
        ...p,
        yellowCard: true,
        redCard: true,
        onField: false
      };
    } else {
      // First yellow card
      return {
        ...p,
        yellowCard: true
      };
    }
  });
  
  // Log the card
  const cardLog = {
    type: 'warning',
    text: player.yellowCard 
      ? `Second yellow card for ${player.name}! Red card - sent off!`
      : `Yellow card issued to ${player.name}`
  };
  
  // Add to fouls list
  const newFouls = [...gameState.fouls, {
    type: player.yellowCard ? 'second yellow' : 'yellow',
    player: player.id,
    team,
    minute: gameState.matchMinute,
    location: { col, row }
  }];
  
  return {
    ...gameState,
    players: newPlayers,
    gameLog: [...gameState.gameLog, cardLog],
    fouls: newFouls
  };
}

 * Issue a red card to a random player on the specified team

/**
 * Cause an injury to a player
 * @param {Object} gameState Current game state
 * @param {String} team Team of the player to injure
 * @param {Number} col Column location
 * @param {Number} row Row location
 * @returns {Object} Updated game state
 */
function causeInjury(gameState, team, col, row) {
  // Find players from the team at the location
  const playersAtLocation = gameState.players.filter(
    p => p.team === team && p.onField && p.col === col && p.row === row
  );
  
  if (playersAtLocation.length === 0) return gameState;
  
  // Pick a random player to injure
  const randomIndex = Math.floor(Math.random() * playersAtLocation.length);
  const player = playersAtLocation[randomIndex];
  
  // Determine injury severity
  const severityRoll = Math.random();
  let severity;
  let movementReduction = 0;
  let needsSubstitution = false;
  let removeFromPitch = false;
  
  if (severityRoll < 0.6) {
    // 60% chance of minor injury
    severity = 'minor';
    movementReduction = 1;
  } else if (severityRoll < 0.9) {
    // 30% chance of serious injury
    severity = 'serious';
    movementReduction = 2;
    
    // 50% chance the player needs to be substituted
    if (Math.random() < 0.5) {
      needsSubstitution = true;
    }
  } else {
    // 10% chance of season-ending injury
    severity = 'season_ending';
    removeFromPitch = true;
  }
  
  // Create new players array
  const newPlayers = gameState.players.map(p => {
    if (p.id !== player.id) return p;
    
    return {
      ...p,
      injury: {
        severity,
        turnOccurred: gameState.turnNumber,
        matchMinute: gameState.matchMinute
      },
      movementReduction,
      needsSubstitution,
      onField: !removeFromPitch
    };
  });
  
  // Log the injury
  const injuryLog = {
    type: 'error',
    text: `${player.name} has suffered a ${severity.replace('_', ' ')} injury!`
  };
  
  // Add to injuries list
  const newInjuries = [...gameState.injuries, {
    player: player.id,
    team,
    severity,
    minute: gameState.matchMinute,
    location: { col, row }
  }];
  
  return {
    ...gameState,
    players: newPlayers,
    gameLog: [...gameState.gameLog, injuryLog],
    injuries: newInjuries
  };
}

/**
 * Execute a pass between two players
 * @param {Object} gameState Current game state
 * @param {String} fromPlayerId ID of the player passing the ball
 * @param {String} toPlayerId ID of the player receiving the ball
 * @returns {Object} Updated game state
 */
export function executePass(gameState, fromPlayerId, toPlayerId) {
  const { players, currentTeam, movesRemaining } = gameState;
  
  // Find the players
  const fromPlayer = players.find(p => p.id === fromPlayerId);
  const toPlayer = players.find(p => p.id === toPlayerId);
  
  // Validate the pass
  if (!fromPlayer || !toPlayer) return gameState;
  if (!fromPlayer.hasBall) return gameState;
  if (fromPlayer.team !== toPlayer.team) return gameState;
  if (fromPlayer.team !== currentTeam) return gameState;
  if (!fromPlayer.onField || !toPlayer.onField) return gameState;
  
  // Calculate pass distance
  const distance = Math.sqrt(
    Math.pow(toPlayer.col - fromPlayer.col, 2) + 
    Math.pow(toPlayer.row - fromPlayer.row, 2)
  );
  
  // Check if pass is within range (based on player's passing attribute)
  const maxPassDistance = 5 + (fromPlayer.attributes.passing / 20); // Base 5 plus bonus from attribute
  
  if (distance > maxPassDistance) {
    // Pass is too long, log the failure
    const failLog = {
      type: 'warning',
      text: `${fromPlayer.name}'s pass to ${toPlayer.name} is too long!`
    };
    
    return {
      ...gameState,
      gameLog: [...gameState.gameLog, failLog]
    };
  }
  
  // Check for interception chance
  const intercepted = checkForInterception(gameState, fromPlayer, toPlayer);
  if (intercepted.intercepted) {
    return intercepted.updatedGameState;
  }
  
  // Pass is successful, update possession
  const updatedGameState = updateBallPossession(gameState, toPlayerId);
  
  // Log the successful pass
  const passLog = {
    type: 'action',
    text: `${fromPlayer.name} passes to ${toPlayer.name}`
  };
  
  // Reduce moves remaining
  const finalGameState = {
    ...updatedGameState,
    gameLog: [...updatedGameState.gameLog, passLog],
    movesRemaining: movesRemaining - 1
  };
  
  // End turn if no moves remaining
  if (finalGameState.movesRemaining <= 0) {
    return endTurn(finalGameState);
  }
  
  return finalGameState;
}

/**
 * Check if a pass might be intercepted
 * @param {Object} gameState Current game state
 * @param {Object} fromPlayer Player passing the ball
 * @param {Object} toPlayer Player receiving the ball
 * @returns {Object} Result with intercepted flag and updated state if needed
 */
function checkForInterception(gameState, fromPlayer, toPlayer) {
  // Find opposing players who might be in the passing lane
  const opposingTeam = fromPlayer.team === TEAMS.HOME ? TEAMS.AWAY : TEAMS.HOME;
  const possibleInterceptors = gameState.players.filter(p => 
    p.team === opposingTeam && 
    p.onField && 
    isInPassingLane(fromPlayer, toPlayer, p)
  );
  
  if (possibleInterceptors.length === 0) {
    return { intercepted: false };
  }
  
  // For each potential interceptor, calculate chance based on attributes
  for (const interceptor of possibleInterceptors) {
    const passingPower = fromPlayer.attributes.passing;
    const interceptorSkill = interceptor.attributes.defending;
    
    // Base interception chance increases with defender skill and decreases with passer skill
    let interceptionChance = 0.3 + (interceptorSkill / 200) - (passingPower / 200);
    
    // Clamp between 5% and 60%
    interceptionChance = Math.max(0.05, Math.min(0.6, interceptionChance));
    
    if (Math.random() < interceptionChance) {
      // Pass is intercepted!
      const interceptLog = {
        type: 'warning',
        text: `${interceptor.name} intercepts the pass from ${fromPlayer.name}!`
      };
      
      // Update possession to the interceptor
      const updatedGameState = updateBallPossession(gameState, interceptor.id);
      
      return {
        intercepted: true,
        updatedGameState: {
          ...updatedGameState,
          gameLog: [...updatedGameState.gameLog, interceptLog],
          movesRemaining: updatedGameState.movesRemaining - 1
        }
      };
    }
  }
  
  return { intercepted: false };
}

/**
 * Check if a player is in the passing lane between two players
 * @param {Object} fromPlayer Starting player
 * @param {Object} toPlayer Target player
 * @param {Object} interceptor Potential intercepting player
 * @returns {Boolean} True if the player is in the passing lane
 */
function isInPassingLane(fromPlayer, toPlayer, interceptor) {
  // Calculate distance between the two players
  const passDistance = Math.sqrt(
    Math.pow(toPlayer.col - fromPlayer.col, 2) + 
    Math.pow(toPlayer.row - fromPlayer.row, 2)
  );
  
  // Calculate distances from interceptor to both players
  const distToFrom = Math.sqrt(
    Math.pow(interceptor.col - fromPlayer.col, 2) + 
    Math.pow(interceptor.row - fromPlayer.row, 2)
  );
  
  const distToTo = Math.sqrt(
    Math.pow(interceptor.col - toPlayer.col, 2) + 
    Math.pow(interceptor.row - toPlayer.row, 2)
  );
  
  // If sum of distances is roughly equal to the pass distance, the player is in the lane
  // Allow a small margin for error to account for grid positioning
  const tolerance = 1.2;
  return (distToFrom + distToTo) < (passDistance + tolerance);
}

/**
 * Execute a shot at goal
 * @param {Object} gameState Current game state
 * @param {String} playerId ID of the player shooting
 * @returns {Object} Updated game state
 */
export function executeShot(gameState, playerId) {
  const { players, currentTeam } = gameState;
  
  // Find the player
  const player = players.find(p => p.id === playerId);
  
  // Validate the shot
  if (!player || !player.hasBall) return gameState;
  if (player.team !== currentTeam) return gameState;
  if (!player.onField) return gameState;
  
  // Check if player is in shooting range
  const inShootingRange = isInShootingRange(player);
  
  if (!inShootingRange) {
    // Too far to shoot
    const rangeLog = {
      type: 'warning',
      text: `${player.name} is too far from goal to shoot!`
    };
    
    return {
      ...gameState,
      gameLog: [...gameState.gameLog, rangeLog]
    };
  }
  
  // Determine which goal to shoot at
  const targetGoalCol = player.team === TEAMS.HOME ? 19 : 0;
  const targetGoalRows = [4, 5, 6]; // The three squares that form the goal
  
  // Choose a random target within the goal
  const targetRow = targetGoalRows[Math.floor(Math.random() * targetGoalRows.length)];
  
  // Check for defenders and goalkeeper in the way
  const defendingTeam = player.team === TEAMS.HOME ? TEAMS.AWAY : TEAMS.HOME;
  const defendersInGoal = players.filter(p => 
    p.team === defendingTeam && 
    p.col === targetGoalCol && 
    targetGoalRows.includes(p.row) &&
    p.onField
  );
  
  const goalkeepersInTarget = defendersInGoal.filter(p => p.position === 'GK');
  const totalDefenders = defendersInGoal.length;
  
  // Calculate shot success probability
  let successProbability = 0.5; // Base 50% chance
  
  // Increase based on shooting attribute
  successProbability += (player.attributes.shooting - 70) / 100;
  
  // Reduce by 15% for each defender
  successProbability -= totalDefenders * 0.15;
  
  // Reduce by 25% if goalkeeper is positioned correctly
  if (goalkeepersInTarget.some(gk => gk.row === targetRow)) {
    const goalkeeper = goalkeepersInTarget.find(gk => gk.row === targetRow);
    successProbability -= 0.25 + (goalkeeper.attributes.defending - 70) / 200;
  }
  
  // Minimum 5% chance to score, maximum 95%
  successProbability = Math.max(0.05, Math.min(0.95, successProbability));
  
  // Attempt the shot
  const shotResult = Math.random() < successProbability;
  
  if (shotResult) {
    // Goal scored!
    return handleGoalScored(gameState, player);
  } else {
    // Shot saved or missed
    return handleShotMissed(gameState, player, targetRow, defendingTeam, goalkeepersInTarget);
  }
}

/**
 * Check if a player is in shooting range
 * @param {Object} player Player to check
 * @returns {Boolean} True if player is in shooting range
 */
function isInShootingRange(player) {
  if (player.team === TEAMS.HOME) {
    return player.col >= 16; // Within 4 columns of away goal
  } else {
    return player.col <= 3; // Within 4 columns of home goal
  }
}

/**
 * Handle a successful goal
 * @param {Object} gameState Current game state
 * @param {Object} player Player who scored
 * @returns {Object} Updated game state
 */
function handleGoalScored(gameState, player) {
  // Update score
  const newScore = { ...gameState.score };
  if (player.team === TEAMS.HOME) {
    newScore.home += 1;
  } else {
    newScore.away += 1;
  }
  
  // Log the goal
  const goalLog = {
    type: 'goal',
    text: `GOAL! ${player.name} scores for the ${player.team} team!`
  };
  
  // Reset positions for kickoff (defending team will kick off)
  const defendingTeam = player.team === TEAMS.HOME ? TEAMS.AWAY : TEAMS.HOME;
  
  return {
    ...gameState,
    score: newScore,
    gameLog: [...gameState.gameLog, goalLog],
    gameState: GAME_STATES.GOAL_SCORED,
    // The reset positions logic would be implemented separately
    // as it involves a more complex state transition
  };
}

/**
 * Handle a missed or saved shot
 * @param {Object} gameState Current game state
 * @param {Object} player Player who shot
 * @param {Number} targetRow Row that was targeted
 * @param {String} defendingTeam Team that was defending
 * @param {Array} goalkeepers Goalkeepers in the goal area
 * @returns {Object} Updated game state
 */
function handleShotMissed(gameState, player, targetRow, defendingTeam, goalkeepers) {
  // Determine if it was saved or missed
  const isSaved = goalkeepers.length > 0;
  
  // Log the result
  const resultLog = {
    type: 'action',
    text: isSaved 
      ? `${goalkeepers[0].name} saves the shot from ${player.name}!`
      : `${player.name}'s shot misses the target!`
  };
  
  // Give ball to goalkeeper if saved
  let updatedGameState = gameState;
  
  if (isSaved) {
    const goalkeeper = goalkeepers[0];
    updatedGameState = updateBallPossession(gameState, goalkeeper.id);
  } else {
    // Ball goes out for a goal kick
    // In a real implementation, we would set up for a goal kick
    // For this example, just remove ball possession from all players
    const newPlayers = gameState.players.map(p => ({
      ...p,
      hasBall: false
    }));
    
    updatedGameState = {
      ...gameState,
      players: newPlayers,
      ball: {
        ...gameState.ball,
        possessionPlayerId: null
      }
    };
  }
  
  // End turn after a shot
  return endTurn({
    ...updatedGameState,
    gameLog: [...updatedGameState.gameLog, resultLog]
  });
}

/**
 * End the current turn and switch teams
 * @param {Object} gameState Current game state
 * @returns {Object} Updated game state
 */
export function endTurn(gameState) {
  const newTeam = gameState.currentTeam === TEAMS.HOME ? TEAMS.AWAY : TEAMS.HOME;
  
  // Log turn change
  const turnLog = {
    type: 'info',
    text: `${newTeam.charAt(0).toUpperCase() + newTeam.slice(1)} team's turn`
  };
  
  // Increment match minute
  const newMatchMinute = gameState.matchMinute + 5; // Each turn represents 5 minutes
  
  // Check for half-time or full-time
  let newGameState = gameState.gameState;
  
  if (newMatchMinute === 45) {
    newGameState = GAME_STATES.HALF_TIME;
    
    // Log half-time
    turnLog.text = "Half-time! Teams switch sides.";
  } else if (newMatchMinute >= 90) {
    newGameState = GAME_STATES.FULL_TIME;
    
    // Log full-time
    turnLog.text = "Full-time! The match is over.";
  }
  
  return {
    ...gameState,
    currentTeam: newTeam,
    movesRemaining: MOVES_PER_TURN,
    turnNumber: gameState.turnNumber + 1,
    matchMinute: newMatchMinute,
    gameState: newGameState,
    gameLog: [...gameState.gameLog, turnLog]
  };
}
