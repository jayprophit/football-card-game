      if (!this.web3Service || !this.web3Service.walletConnected) {
        throw new Error('Web3 service not initialized or wallet not connected');
      }
      
      // Cancel listing
      const result = await this.web3Service.cancelMarketplaceListing(listingId);
      
      // Update user listings
      if (this.userListings.has(userId)) {
        const listings = this.userListings.get(userId);
        const index = listings.findIndex(l => l.listingId === listingId);
        
        if (index >= 0) {
          listings.splice(index, 1);
        }
      }
      
      return result;
    } catch (error) {
      console.error('Error cancelling listing:', error);
      throw error;
    }
  }

  getUserListings(userId) {
    return this.userListings.get(userId) || [];
  }

  async checkItemAgeRestriction(listingId, userAgeVerification) {
    // Get listing from contract or cache
    const listing = this.activeListings.find(l => l.id === listingId);
    
    if (!listing) {
      throw new Error('Listing not found');
    }
    
    // Check if item is age-restricted
    let minimumAge = 0;
    
    if (listing.isPlayerCard) {
      // Player cards might have restrictions based on rarity
      if (listing.item && listing.item.legendary) {
        minimumAge = 18; // Legendary cards require 18+
      } else {
        minimumAge = 13; // Regular cards require 13+
      }
    } else if (listing.isTeam) {
      // Teams might have restrictions
      minimumAge = 16; // Teams require 16+
    }
    
    return {
      isRestricted: minimumAge > 0,
      minimumAge,
      userMeetsRequirement: userAgeVerification.age >= minimumAge
    };
  }

/**

 */
class TournamentManager {
  constructor() {
    this.tournaments = new Map();
    this.userEntries = new Map();
    this.web3Service = null;
    this.blockchainTournaments = null;
  }
  
  // Initialize tournament system
  initialize(web3Service) {
    this.web3Service = web3Service;
    
    if (web3Service && web3Service.contracts.tournament) {
      this.blockchainTournaments = new TournamentSystem(web3Service);
      this.blockchainTournaments.initialize();
    }
  }
  
  // Create tournament
  async createTournament(options) {
    // Generate tournament ID
    const tournamentId = `tournament-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
    
    // Create tournament object
    const tournament = {
      id: tournamentId,
      name: options.name,
      description: options.description,
      createdBy: options.createdBy,
      startDate: options.startDate,
      endDate: options.endDate,
      maxParticipants: options.maxParticipants || 32,
      participants: [],
      matches: [],
      prizes: options.prizes || [],
      entryFee: options.entryFee || 0,
      status: 'created',
      requiresAge: options.requiresAge || 16,
      requiresKyc: options.requiresKyc || false,
      onBlockchain: options.onBlockchain || false,
      blockchainTournamentId: null,
      createdAt: new Date()
    };
    
    // Store tournament
    this.tournaments.set(tournamentId, tournament);
    
    // Create blockchain tournament if required
    if (options.onBlockchain && this.blockchainTournaments) {
      try {
        const blockchainResult = await this.blockchainTournaments.createTournament({
          name: options.name,
          description: options.description,
          entryFee: options.entryFee || 0,
          maxParticipants: options.maxParticipants || 32,
          startTime: new Date(options.startDate),
          endTime: new Date(options.endDate),
          tournamentType: options.type || 'knockout',
          nftRequired: options.requiresNft || false
        });
        
        tournament.blockchainTournamentId = blockchainResult.tournamentId;
        tournament.blockchainTransactionHash = blockchainResult.transactionHash;
      } catch (error) {
        console.error('Error creating blockchain tournament:', error);
        tournament.blockchainError = error.message;
      }
    }
    
    return tournament;
  }
  
  // Get tournament
  getTournament(tournamentId) {
    return this.tournaments.get(tournamentId) || null;
  }
  
  // Get all tournaments
  getAllTournaments() {
    return Array.from(this.tournaments.values());
  }
  
  // Get active tournaments
  getActiveTournaments() {
    const now = new Date();
    return Array.from(this.tournaments.values()).filter(t => 
      new Date(t.startDate) <= now && new Date(t.endDate) >= now && t.status !== 'completed'
    );
  }
  
  // Enter tournament
  async enterTournament(userId, tournamentId) {
    const tournament = this.tournaments.get(tournamentId);
    
    if (!tournament) {
      throw new Error('Tournament not found');
    }
    
    // Check if tournament is full
    if (tournament.participants.length >= tournament.maxParticipants) {
      throw new Error('Tournament is full');
    }
    
    // Check if user already entered
    if (tournament.participants.includes(userId)) {
      throw new Error('User already entered tournament');
    }
    
    // Add user to participants
    tournament.participants.push(userId);
    
    // Record entry
    if (!this.userEntries.has(userId)) {
      this.userEntries.set(userId, []);
    }
    
    const entry = {
      userId,
      tournamentId,
      enteredAt: new Date(),
      status: 'active'
    };
    
    this.userEntries.get(userId).push(entry);
    
    // Enter blockchain tournament if applicable
    if (tournament.onBlockchain && tournament.blockchainTournamentId && this.blockchainTournaments) {
      try {
        // For blockchain tournaments, user needs to have a team
        // This would be implemented with proper team selection
        const teamId = 1; // Placeholder
        
        const blockchainResult = await this.blockchainTournaments.registerForTournament(
          tournament.blockchainTournamentId,
          teamId
        );
        
        entry.blockchainRegistered = true;
        entry.blockchainTransactionHash = blockchainResult.transactionHash;
      } catch (error) {
        console.error('Error entering blockchain tournament:', error);
        entry.blockchainError = error.message;
      }
    }
    
    return entry;
  }
  
  // Get user tournament entries
  getUserTournamentEntries(userId) {
    return this.userEntries.get(userId) || [];
  }
  
  // Check if user meets tournament requirements
  checkTournamentRequirements(userId, tournamentId, userProfile) {
    const tournament = this.tournaments.get(tournamentId);
    
    if (!tournament) {
      throw new Error('Tournament not found');
    }
    
    // Check age requirement
    const meetsAgeRequirement = userProfile.age >= tournament.requiresAge;
    
    // Check KYC requirement
    const meetsKycRequirement = !tournament.requiresKyc || 
      (userProfile.kycStatus && userProfile.kycStatus.status === 'approved');
    
    return {
      meetsRequirements: meetsAgeRequirement && meetsKycRequirement,
      meetsAgeRequirement,
      meetsKycRequirement,
      minimumAge: tournament.requiresAge,
      requiresKyc: tournament.requiresKyc
    };
  }
}

 */
class PlayerProgressionSystem {
  constructor() {
    this.playerProgress = new Map();
    this.playerCards = new Map();
    this.levelRequirements = [
      100,   // Level 1 -> 2
      300,   // Level 2 -> 3
      600,   // Level 3 -> 4
      1000,  // Level 4 -> 5
      1500,  // Level 5 -> 6
      2100,  // Level 6 -> 7

      3600,  // Level 8 -> 9

      5500,  // Level 10 -> 11
      6600,  // Level 11 -> 12
      7800,  // Level 12 -> 13
      9100,  // Level 13 -> 14
      10500, // Level 14 -> 15
      12000  // Level 15 -> 16
    ];
    this.levelRewards = [
      { coins: 100, packType: 'bronze' },     // Level 2
      { coins: 200, packType: 'bronze' },     // Level 3
      { coins: 300, packType: 'silver' },     // Level 4
      { coins: 400, packType: 'silver' },     // Level 5
      { coins: 500, packType: 'gold' },       // Level 6
      { coins: 600, packType: 'gold' },       // Level 7
      { coins: 700, packType: 'premium' },    // Level 8
      { coins: 800, packType: 'premium' },    // Level 9
      { coins: 900, packType: 'elite' },      // Level 10
      { coins: 1000, packType: 'elite' },     // Level 11
      { coins: 1100, packType: 'legendary' }, // Level 12
      { coins: 1200, packType: 'legendary' }, // Level 13
      { coins: 1300, packType: 'ultimate' },  // Level 14
      { coins: 1400, packType: 'ultimate' },  // Level 15
      { coins: 1500, packType: 'ultimate' }   // Level 16
    ];
  }
  
  // Initialize player progression
  initialize(userId) {
    // Create progression object if it doesn't exist
    if (!this.playerProgress.has(userId)) {
      this.playerProgress.set(userId, {
        userId,
        level: 1,
        experience: 0,
        coins: 0,
        matches: {
          played: 0,
          won: 0,
          drawn: 0,
          lost: 0
        },
        stats: {
          goalsScored: 0,
          goalsConceded: 0,
          cleanSheets: 0,
          cards: {
            yellow: 0,
            red: 0
          }
        },
        rewards: [],
        createdAt: new Date(),
        updatedAt: new Date()
      });
    }
    
    // Initialize cards collection if it doesn't exist
    if (!this.playerCards.has(userId)) {
      this.playerCards.set(userId, []);
    }
    
    return this.playerProgress.get(userId);
  }
  
  // Get player progress
  getPlayerProgress(userId) {
    return this.playerProgress.get(userId) || null;
  }
  
  // Add experience
  async addExperience(userId, amount, reason) {
    const progress = this.playerProgress.get(userId);
    
    if (!progress) {
      throw new Error('Player progress not found');
    }
    
    // Calculate current level
    const currentLevel = progress.level;
    
    // Add experience
    progress.experience += amount;
    
    // Check if player leveled up
    const newLevel = this._calculateLevel(progress.experience);
    const leveledUp = newLevel > currentLevel;
    
    // Update level if changed
    if (leveledUp) {
      progress.level = newLevel;
      
      // Process level-up rewards
      const levelUps = [];
      
      for (let lvl = currentLevel + 1; lvl <= newLevel; lvl++) {
        const reward = this.levelRewards[lvl - 2]; // -2 because rewards start at level 2
        
        if (reward) {
          progress.coins += reward.coins;
          
          // Record level up reward
          levelUps.push({
            level: lvl,
            reward,
            timestamp: new Date()
          });
          
          // Add to rewards history
          progress.rewards.push({
            type: 'level_up',
            level: lvl,
            coins: reward.coins,
            packType: reward.packType,
            timestamp: new Date()
          });
        }
      }
    }
    
    // Update last update time
    progress.updatedAt = new Date();
    
    // Update in store
    this.playerProgress.set(userId, progress);
    
    return {
      newExperience: progress.experience,
      previousLevel: currentLevel,
      newLevel: progress.level,
      leveledUp,
      levelUps: leveledUp ? progress.rewards.filter(r => r.type === 'level_up' && r.level > currentLevel) : []
    };
  }
  
  // Calculate level based on experience
  _calculateLevel(experience) {
    let level = 1;
    
    for (let i = 0; i < this.levelRequirements.length; i++) {
      if (experience >= this.levelRequirements[i]) {
        level++;
      } else {
        break;
      }
    }
    
    return level;
  }
  
  // Add cards to player's collection
  async addCards(userId, cards) {
    if (!this.playerCards.has(userId)) {
      this.playerCards.set(userId, []);
    }
    
    const userCards = this.playerCards.get(userId);
    
    // Add unique ID to each card and timestamp
    const cardsWithIds = cards.map(card => ({
      ...card,
      id: `card-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
      acquiredAt: new Date()
    }));
    
    // Add to collection
    userCards.push(...cardsWithIds);
    
    // Update in store
    this.playerCards.set(userId, userCards);
    
    return cardsWithIds;
  }
  
  // Get player cards
  getPlayerCards(userId) {
    return this.playerCards.get(userId) || [];
  }
  
  // Add coins
  async addCoins(userId, amount, reason) {
    const progress = this.playerProgress.get(userId);
    
    if (!progress) {
      throw new Error('Player progress not found');
    }
    
    // Add coins
    progress.coins += amount;
    
    // Record reward
    progress.rewards.push({
      type: 'coins',
      amount,
      reason,
      timestamp: new Date()
    });
    
    // Update last update time
    progress.updatedAt = new Date();
    
    // Update in store
    this.playerProgress.set(userId, progress);
    
    return {
      previousCoins: progress.coins - amount,
      newCoins: progress.coins,
      added: amount
    };
  }
  
  // Update match statistics
  async updateMatchStats(userId, matchResult) {
    const progress = this.playerProgress.get(userId);
    
    if (!progress) {
      throw new Error('Player progress not found');
    }
    
    // Update match count
    progress.matches.played++;
    
    // Update result
    if (matchResult.won) {
      progress.matches.won++;
    } else if (matchResult.drawn) {
      progress.matches.drawn++;
    } else {
      progress.matches.lost++;
    }
    
    // Update goals
    progress.stats.goalsScored += matchResult.goalsScored || 0;
    progress.stats.goalsConceded += matchResult.goalsConceded || 0;
    
    // Update clean sheets
    if (matchResult.goalsConceded === 0) {
      progress.stats.cleanSheets++;
    }
    
    // Update cards
    if (matchResult.cards) {
      progress.stats.cards.yellow += matchResult.cards.yellow || 0;
      progress.stats.cards.red += matchResult.cards.red || 0;
    }
    
    // Update last update time
    progress.updatedAt = new Date();
    
    // Update in store
    this.playerProgress.set(userId, progress);
    
    return progress;
  }
}

      }

      for (const reward of pendingRewards) {
        totalAmount += reward.amount;
      }
      
      // Award total amount
      try {
        const result = await this.awardRewards(userId, totalAmount, 'Pending rewards processing');
        
        // Mark all as processed
        for (const reward of pendingRewards) {
          reward.status = 'processed';
          reward.processedAt = new Date();
          
          results.push({
            pendingId: reward.id,
            amount: reward.amount,
            reason: reward.reason,
            success: true
          });
        }
        
        // Clear processed rewards
        this.pendingRewards.set(userId, []);
        
        return {
          success: true,
          totalAmount,
          transactionHash: result.transactionHash,
          processed: pendingRewards.length,
          rewards: results
        };
      } catch (error) {
        console.error('Error processing pending rewards:', error);
        
        // Mark all as failed
        for (const reward of pendingRewards) {
          reward.status = 'failed';
          reward.error = error.message;
          
          results.push({
            pendingId: reward.id,
            amount: reward.amount,
            reason: reward.reason,
            success: false,
            error: error.message
          });
        }
        
        return {
          success: false,
          error: error.message,
          failed: pendingRewards.length,
          rewards: results
        };
      }
    } catch (error) {

      throw error;
    }
  }
  
  // Prepare player card for future minting (for underage users)
  async preparePlayerCard(userId, cardData) {
    try {
      // Generate unique ID for pending NFT
      const pendingId = `pending-${userId}-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
      
      // Store pending NFT data
      const pendingNFT = {
        id: pendingId,
        userId,
        type: 'player_card',
        data: cardData,
        createdAt: new Date(),
        status: 'pending'
      };
      
      // Add to pending NFTs
      if (!this.pendingNFTs.has(userId)) {
        this.pendingNFTs.set(userId, []);
      }
      
      this.pendingNFTs.get(userId).push(pendingNFT);
      
      return {
        success: true,
        pendingId,
        status: 'pending'
      };
    } catch (error) {
      console.error('Error preparing player card:', error);
      throw error;
    }
  }
  
  // Process pending NFTs for a user (e.g., after turning 18 or completing KYC)
  async processPendingNFTs(userId) {
    try {
      if (!this.web3Service || !this.web3Service.walletConnected) {
        throw new Error('Web3 service not initialized or wallet not connected');
      }
      
      // Get pending NFTs for user
      const pendingNFTs = this.pendingNFTs.get(userId) || [];
      
      if (pendingNFTs.length === 0) {
        return { success: true, processed: 0, message: 'No pending NFTs' };
      }
      
      const results = [];
      
      // Process each pending NFT
      for (const pendingNFT of pendingNFTs) {
        try {
          let result;
          
          // Process based on NFT type
          if (pendingNFT.type === 'player_card') {
            result = await this.mintPlayerCard(pendingNFT.data);
          } else if (pendingNFT.type === 'team') {
            result = await this.mintTeam(pendingNFT.data);
          } else {
            throw new Error(`Unknown NFT type: ${pendingNFT.type}`);
          }
          
          // Mark as processed
          pendingNFT.status = 'processed';
          pendingNFT.processedAt = new Date();
          pendingNFT.result = result;
          
          results.push({
            pendingId: pendingNFT.id,
            success: true,
            result
          });
        } catch (error) {
          console.error(`Error processing pending NFT ${pendingNFT.id}:`, error);
          
          // Mark as failed
          pendingNFT.status = 'failed';
          pendingNFT.error = error.message;
          
          results.push({
            pendingId: pendingNFT.id,
            success: false,
            error: error.message
          });
        }
      }
      
      // Update pending NFTs, removing processed ones
      this.pendingNFTs.set(
        userId,
        pendingNFTs.filter(nft => nft.status !== 'processed')
      );
      
      return {
        success: true,
        total: pendingNFTs.length,
        processed: results.filter(r => r.success).length,
        failed: results.filter(r => !r.success).length,
        results
      };
    } catch (error) {
      console.error('Error processing pending NFTs:', error);
      throw error;
    }
  }
  
  // Store a pending NFT for future delivery
  async preparePendingNFT(userId, nftData, unlockDate = null) {
    try {
      // Generate unique ID for pending NFT
      const pendingId = `pending-${userId}-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
      
      // Store pending NFT data
      const pendingNFT = {
        id: pendingId,
        userId,
        type: 'external_nft',
        data: nftData,
        unlockDate,
        createdAt: new Date(),
        status: unlockDate ? 'pending_age' : 'pending'
      };
      
      // Add to pending NFTs
      if (!this.pendingNFTs.has(userId)) {
        this.pendingNFTs.set(userId, []);
      }
      
      this.pendingNFTs.get(userId).push(pendingNFT);
      
      return {
        success: true,
        pendingId,
        status: pendingNFT.status
      };
    } catch (error) {
      console.error('Error preparing pending NFT:', error);
      throw error;
    }
  }
  
  // Get movement type based on position
  _getMovementTypeForPosition(position) {
    // Map football positions to chess-like movement patterns
    const positionMovements = {
      'Goalkeeper': 'Rook',
      'Defender': 'Bishop',
      'Midfielder': 'Queen',
      'Forward': 'Knight',
      'Striker': 'Knight',
      'Winger': 'Knight'
    };
    
    return positionMovements[position] || 'Pawn';
  }

/**

class UserManagementSystem {
  constructor() {
    this.authService = new AuthenticationService();
    this.userProfileService = new UserProfileService();
    this.ageVerificationService = new AgeVerificationService();
    this.kycService = new KYCService();
    this.rewardsLockService = new RewardsLockService();
    this.userPreferences = new UserPreferencesService();
    this.analytics = new UserAnalyticsService();
  }
  
  // Register new user
  async registerUser(username, email, password, dateOfBirth) {
    // Create user account
    const user = await this.authService.registerUser(username, email, password);
    
    // Calculate age
    const age = this._calculateAge(dateOfBirth);
    
    // Create user profile
    await this.userProfileService.createProfile(user.id, {
      username,
      email,
      dateOfBirth,
      age,
      createdAt: new Date(),
      isUnderage: age < 18
    });
    
    // Initialize age verification status
    await this.ageVerificationService.initializeStatus(user.id, dateOfBirth);
    
    // Initialize rewards lock if underage
    if (age < 18) {
      await this.rewardsLockService.initializeUnderageUser(user.id, dateOfBirth);
    }
    
    return user;
  }
  
  // Calculate age from date of birth
  _calculateAge(dateOfBirth) {
    const dob = new Date(dateOfBirth);
    const today = new Date();
    let age = today.getFullYear() - dob.getFullYear();
    const monthDiff = today.getMonth() - dob.getMonth();
    
    if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < dob.getDate())) {
      age--;
    }
    
    return age;
  }
  
  // Login user
  async loginUser(username, password) {
    // Authenticate user
    const authResult = await this.authService.loginUser(username, password);
    
    // Get user profile
    const profile = await this.userProfileService.getProfile(authResult.user.id);
    
    // Get age verification status
    const ageStatus = await this.ageVerificationService.getStatus(authResult.user.id);
    
    // Get KYC status if applicable
    let kycStatus = null;
    if (profile.age >= 18) {
      kycStatus = await this.kycService.getStatus(authResult.user.id);
    }
    
    // Get rewards status
    const rewardsStatus = await this.rewardsLockService.getStatus(authResult.user.id);
    
    // Record login for analytics
    this.analytics.recordLogin(authResult.user.id);
    
    return {
      ...authResult,
      profile,
      ageStatus,
      kycStatus,
      rewardsStatus
    };
  }
  
  // Update user's KYC status
  async updateKYC(userId, kycData, documents) {

    if (kycResult.status === 'approved') {

      },

    };

    };
    
    const requiredAge = ageRestrictions[feature] || 0;
    
    return {
      allowed: status.age >= requiredAge,
      requiredAge,
      currentAge: status.age
    };
  }
  
  // Update age status (e.g., on birthday)
  async updateAgeStatus(userId) {
    const status = this.ageStatuses.get(userId);
    
    if (!status) {
      throw new Error('User age verification status not found');
    }
    
    // Recalculate age
    const today = new Date();
    let age = today.getFullYear() - status.dateOfBirth.getFullYear();
    const monthDiff = today.getMonth() - status.dateOfBirth.getMonth();
    
    if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < status.dateOfBirth.getDate())) {
      age--;
    }
    
    // Update status
    status.age = age;
    status.isUnderage = age < 18;
    status.lastChecked = today;
    
    // Clear unlock date if now 18+
    if (age >= 18) {
      status.isUnderage = false;
    }
    
    // Update stored status
    this.ageStatuses.set(userId, status);
    
    return status;
  }
}

/**
 * Rewards Lock Service
 * Handles locking and unlocking rewards for underage users
 */
class RewardsLockService {
  constructor() {
    this.lockedRewards = new Map(); // User ID to locked rewards
    this.stakingYield = 0.08; // 8% annual yield
    this.compoundingFrequency = 'daily'; // Compounding frequency
  }
  
  // Initialize underage user
  async initializeUnderageUser(userId, dateOfBirth) {
    // Calculate unlock date (18th birthday)
    const unlockDate = new Date(dateOfBirth);
    unlockDate.setFullYear(unlockDate.getFullYear() + 18);
    
    // Create locked rewards record
    const lockedReward = {
      userId,
      locked: 0, // Initial locked amount
      stakingYield: this.stakingYield,
      compoundingFrequency: this.compoundingFrequency,
      unlockDate,
      projectedValue: 0,
      transactions: [],
      createdAt: new Date(),
      updatedAt: new Date()
    };
    
    // Store record
    this.lockedRewards.set(userId, lockedReward);
    
    return lockedReward;
  }
  
  // Get locked rewards status
  async getStatus(userId) {
    return this.lockedRewards.get(userId) || null;
  }
  
  // Lock rewards for underage user
  async lockRewards(userId, amount, reason) {
    const lockedReward = this.lockedRewards.get(userId);
    
    if (!lockedReward) {
      throw new Error('Locked rewards record not found');
    }
    
    // Add to locked amount
    lockedReward.locked += amount;
    
    // Record transaction
    lockedReward.transactions.push({
      type: 'lock',
      amount,
      reason,
      timestamp: new Date()
    });
    
    // Update projected value
    this._updateProjectedValue(lockedReward);
    
    // Update record
    lockedReward.updatedAt = new Date();
    this.lockedRewards.set(userId, lockedReward);
    
    return lockedReward;
  }
  
  // Update projected value based on staking yield
  _updateProjectedValue(lockedReward) {
    // Calculate time until unlock in years
    const now = new Date();
    const yearsUntilUnlock = (lockedReward.unlockDate - now) / (365.25 * 24 * 60 * 60 * 1000);
    
    if (yearsUntilUnlock <= 0) {
      // User is now 18 or older
      lockedReward.projectedValue = lockedReward.locked;
      return;
    }
    
    // Calculate compound interest
    const principal = lockedReward.locked;
    const rate = lockedReward.stakingYield;
    
    // Convert compounding frequency to number of times per year
    let compoundsPerYear;
    switch (lockedReward.compoundingFrequency) {
      case 'daily':
        compoundsPerYear = 365;
        break;
      case 'weekly':
        compoundsPerYear = 52;
        break;
      case 'monthly':
        compoundsPerYear = 12;
        break;
      case 'quarterly':
        compoundsPerYear = 4;
        break;
      default:
        compoundsPerYear = 1; // Annual
    }
    
    // Compound interest formula: A = P(1 + r/n)^(nt)
    const compoundedAmount = principal * Math.pow(
      (1 + rate / compoundsPerYear),
      compoundsPerYear * yearsUntilUnlock
    );
    
    lockedReward.projectedValue = compoundedAmount;
  }
  
  // Unlock rewards when user turns 18
  async unlockUserRewards(userId) {
    const lockedReward = this.lockedRewards.get(userId);
    
    if (!lockedReward) {
      throw new Error('Locked rewards record not found');
    }
    
    // Update projected value one last time
    this._updateProjectedValue(lockedReward);
    
    // Get final amount to unlock
    const amountToUnlock = lockedReward.projectedValue;
    
    // Record transaction
    lockedReward.transactions.push({
      type: 'unlock',
      amount: amountToUnlock,
      reason: 'Age restriction lifted',
      timestamp: new Date()
    });
    
    // Reset locked amount
    lockedReward.locked = 0;
    lockedReward.projectedValue = 0;
    
    // Update record
    lockedReward.updatedAt = new Date();
    this.lockedRewards.set(userId, lockedReward);
    
    return {
      userId,
      unlockedAmount: amountToUnlock,
      transaction: lockedReward.transactions[lockedReward.transactions.length - 1]
    };
  }
  
  // Get locked rewards history
  async getLockedRewardsHistory(userId) {
    const lockedReward = this.lockedRewards.get(userId);
    
    if (!lockedReward) {
      throw new Error('Locked rewards record not found');
    }
    
    return {
      userId,
      locked: lockedReward.locked,
      projectedValue: lockedReward.projectedValue,
      unlockDate: lockedReward.unlockDate,
      transactions: lockedReward.transactions,
      yield: lockedReward.stakingYield * 100 + '%',
      compoundingFrequency: lockedReward.compoundingFrequency
    };
  }
}

    this.redCards = new Map(); // Player ID to boolean

  }
  
  // Process a foul
  processFoul(player, foulType, inScoringPosition) {
    // Determine card color based on foul type and position
    let cardColor = null;
    
    if (foulType === 'soft') {
      cardColor = inScoringPosition ? 'yellow' : 'warning';
    } else if (foulType === 'hard') {
      cardColor = inScoringPosition ? 'red' : 'yellow';
    } else if (foulType === 'injury') {
      cardColor = 'red';
    }
    
    // Issue card
    if (cardColor === 'yellow') {
      this.issueYellowCard(player);
    } else if (cardColor === 'red') {
      this.issueRedCard(player);
    }
    
    // Record foul
    this.fouls.push({
      player,
      foulType,
      inScoringPosition,
      cardColor,
      timestamp: new Date()
    });
    
    return { cardColor, ejected: cardColor === 'red' };
  }
  
  // Issue yellow card
  issueYellowCard(player) {
    const currentCount = this.yellowCards.get(player.id) || 0;
    this.yellowCards.set(player.id, currentCount + 1);
    
    // Check for second yellow -> red
    if (currentCount + 1 >= 2) {
      this.issueRedCard(player);
      return { ejected: true };
    }
    
    return { ejected: false };
  }
  
  // Issue red card
  issueRedCard(player) {
    this.redCards.set(player.id, true);
    return { ejected: true };
  }
}
